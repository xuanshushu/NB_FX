    static Color32[] Capture(Camera cam, RenderTexture rt, Texture2D readback, string name)
    {
        cam.Render();
        var prev = RenderTexture.active;
        RenderTexture.active = rt;
        readback.ReadPixels(new Rect(0, 0, 128, 128), 0, 0);
        readback.Apply(false);
        RenderTexture.active = prev;
        var bytes = readback.EncodeToPNG();
        File.WriteAllBytes("/tmp/nbfx-t06-20260930/" + name + ".png", bytes);
        return readback.GetPixels32();
    }
    static int Changed(Color32[] a, Color32[] b)
    {
        int n = 0;
        for (int i = 0; i < a.Length; ++i)
            if (Math.Abs(a[i].r-b[i].r)>2 || Math.Abs(a[i].g-b[i].g)>2 || Math.Abs(a[i].b-b[i].b)>2) ++n;
        return n;
    }

            if(Changed(uvBase,uvRot)==0||Changed(uvBase,uvST)==0||Changed(uvBase,uvScroll)==0||Changed(uvBase,uvRotSpeed)==0)throw new Exception("Graph BaseMap UV transform did not affect image");
            // AlphaAll is a late alpha multiplier. Transparent blending must
            // respond; Flags0 bit29 RGB multiplication must occur before it.
            frontMat.SetTexture("_BaseMap",Texture2D.whiteTexture);
            frontMat.SetVector("_BaseMap_ST",new Vector4(1,1,0,0));
            frontMat.SetFloat("_BaseMapUVRotation",0);
            frontMat.SetFloat("_BaseMapUVRotationSpeed",0);
            frontMat.SetVector("_BaseMapMaskMapOffset",Vector4.zero);
            frontMat.SetFloat("_Mask_Toggle",0);
            frontMat.SetFloat("_Mask2_Toggle",0);
            frontMat.SetFloat("_Mask3_Toggle",0);
            frontMat.SetFloat("_Dissolve_Toggle",0);
            frontMat.SetFloat("_NB_Flags0Lo16",0);
            frontMat.SetFloat("_NB_Flags0Hi16",0);
            frontMat.SetFloat("_NB_Flags1Lo16",0);
            frontMat.SetColor("_Color",new Color(1,1,1,0.8f));
            frontMat.SetFloat("_AlphaAll",1);
            var alphaAllOne=Capture(cam,rt,tex,"t07_alphaall_one");
            frontMat.SetFloat("_AlphaAll",0.5f);
            var alphaAllHalf=Capture(cam,rt,tex,"t07_alphaall_half");
            frontMat.SetFloat("_AlphaAll",0);
            var alphaAllZero=Capture(cam,rt,tex,"t07_alphaall_zero");
            Debug.Log("NBFX_T07_ALPHAALL transparent oneHalf="+Changed(alphaAllOne,alphaAllHalf)+" halfZero="+Changed(alphaAllHalf,alphaAllZero));
            if(Changed(alphaAllOne,alphaAllHalf)==0 || Changed(alphaAllHalf,alphaAllZero)==0)throw new Exception("AlphaAll transparent blend inactive");
            frontMat.SetFloat("_Surface",0);
            frontMat.SetFloat("_SrcBlend",(float)BlendMode.One);
            frontMat.SetFloat("_DstBlend",(float)BlendMode.Zero);
            frontMat.SetFloat("_SrcBlendAlpha",(float)BlendMode.One);
            frontMat.SetFloat("_DstBlendAlpha",(float)BlendMode.Zero);
            frontMat.SetFloat("_ZWrite",1);
            frontMat.DisableKeyword("_SURFACE_TYPE_TRANSPARENT");
            frontMat.renderQueue=2000;
            frontMat.SetFloat("_NB_Flags0Hi16",8192);
            frontMat.SetFloat("_AlphaAll",1);
            var premulBeforeOne=Capture(cam,rt,tex,"t07_alphaall_bit29_one");
            frontMat.SetFloat("_AlphaAll",0);
            var premulBeforeZero=Capture(cam,rt,tex,"t07_alphaall_bit29_zero");
            Debug.Log("NBFX_T07_ALPHAALL bit29RgbDiff="+Changed(premulBeforeOne,premulBeforeZero));
            if(Changed(premulBeforeOne,premulBeforeZero)!=0)throw new Exception("AlphaAll ran before bit29 RGB multiplication");
            // ShaderLab's late _ColorA multiplies RGB and alpha before bit29;
            // with opaque One/Zero, alpha alone cannot change RGB unless bit29
            // is on, making ordering directly observable in RGB readback.
            frontMat.SetFloat("_AlphaAll",1);
            frontMat.SetFloat("_NB_Flags0Hi16",0);
            frontMat.SetColor("_ColorA",Color.white);
            var colorABase=Capture(cam,rt,tex,"t07_colora_base");
            frontMat.SetColor("_ColorA",new Color(0.5f,1,1,1));
            var colorARed=Capture(cam,rt,tex,"t07_colora_red_half");
            frontMat.SetColor("_ColorA",new Color(1,1,1,0.5f));
            var colorAAlpha=Capture(cam,rt,tex,"t07_colora_alpha_half_no_bit29");
            frontMat.SetFloat("_NB_Flags0Hi16",8192);
            var colorAAlphaBit29=Capture(cam,rt,tex,"t07_colora_alpha_half_bit29");
            Debug.Log("NBFX_T07_COLORA redDiff="+Changed(colorABase,colorARed)+" alphaWithoutBit29="+Changed(colorABase,colorAAlpha)+" alphaWithBit29="+Changed(colorAAlpha,colorAAlphaBit29));
            if(Changed(colorABase,colorARed)==0 || Changed(colorABase,colorAAlpha)!=0 || Changed(colorAAlpha,colorAAlphaBit29)==0)throw new Exception("ColorA RGB/alpha or bit29 order mismatch");
            // Vertex Color is multiplied just before ColorA unless Flags1 bit9 says ignore.
            frontMat.SetFloat("_NB_Flags0Hi16",0);
            frontMat.SetFloat("_NB_Flags1Lo16",0);
            frontMat.SetColor("_ColorA",Color.white);
            var primitiveDefault = Capture(cam,rt,tex,"t07_vcolor_mesh_default");
            var meshFilter=front.GetComponent<MeshFilter>();
            var coloredMesh=UnityEngine.Object.Instantiate(meshFilter.sharedMesh);
            meshFilter.sharedMesh=coloredMesh;
            var vertexColors=new Color32[coloredMesh.vertexCount];
            for(int i=0;i<vertexColors.Length;++i)vertexColors[i]=new Color32(128,255,255,255);
            coloredMesh.colors32=vertexColors;
            var redHalf=Capture(cam,rt,tex,"t07_vcolor_red_half");
            frontMat.SetFloat("_NB_Flags1Lo16",512);
            var redIgnored=Capture(cam,rt,tex,"t07_vcolor_red_ignored");
            frontMat.SetFloat("_NB_Flags1Lo16",0);
            for(int i=0;i<vertexColors.Length;++i)vertexColors[i]=new Color32(255,255,255,128);
            coloredMesh.colors32=vertexColors;
            var alphaHalf=Capture(cam,rt,tex,"t07_vcolor_alpha_half_no_bit29");
            frontMat.SetFloat("_NB_Flags0Hi16",8192);
            var alphaWithBit29=Capture(cam,rt,tex,"t07_vcolor_alpha_half_bit29");
            int redDiff=Changed(primitiveDefault,redHalf);
            int ignoreDiff=Changed(primitiveDefault,redIgnored);
            int alphaNoBitDiff=Changed(primitiveDefault,alphaHalf);
            int alphaBitDiff=Changed(alphaHalf,alphaWithBit29);
            Debug.Log("NBFX_T07_VERTEX_COLOR redDiff="+redDiff+" ignoredDiff="+ignoreDiff+" alphaWithoutBit29="+alphaNoBitDiff+" alphaWithBit29="+alphaBitDiff);
            if(redDiff==0 || ignoreDiff!=0 || alphaNoBitDiff!=0 || alphaBitDiff==0)
                throw new Exception("Vertex Color or Flags1 bit9/bit29 order mismatch");
            UnityEngine.Object.DestroyImmediate(coloredMesh);
            Debug.Log("NBFX_T06_RESULT visible="+Changed(bg,mode0)+" deferred="+Changed(mode0,deferred)+" camera="+Changed(mode0,camera)+
              " bgCenter="+bg[64*128+64].r+","+bg[64*128+64].g+","+bg[64*128+64].b+
              " mode0Center="+mode0[64*128+64].r+","+mode0[64*128+64].g+","+mode0[64*128+64].b);
        } finally {
            cam.targetTexture=null; rt.Release();
            UnityEngine.Object.DestroyImmediate(rt); UnityEngine.Object.DestroyImmediate(tex);
            UnityEngine.Object.DestroyImmediate(gradient); UnityEngine.Object.DestroyImmediate(maskTexture); UnityEngine.Object.DestroyImmediate(dissolveTexture); UnityEngine.Object.DestroyImmediate(backMat); UnityEngine.Object.DestroyImmediate(frontMat);
            UnityEngine.Object.DestroyImmediate(back); UnityEngine.Object.DestroyImmediate(front); UnityEngine.Object.DestroyImmediate(camGo);
        }
    }
}

internal sealed class NBFXT06MaskCaptureFeature : ScriptableRendererFeature
{
    public Material material;
    Pass pass;
    public override void Create() { pass=new Pass(material) {renderPassEvent=RenderPassEvent.AfterRenderingTransparents+1}; }
    public override void AddRenderPasses(ScriptableRenderer renderer, ref RenderingData renderingData)
    { if (renderingData.cameraData.cameraType==CameraType.Game) renderer.EnqueuePass(pass); }
    sealed class Pass : ScriptableRenderPass
    {
        static readonly int MaskId=Shader.PropertyToID("_DisturbanceMaskTex");
        readonly Material material;
        sealed class PassData { public Material material; }
        public Pass(Material material) { this.material=material; }
        public override void RecordRenderGraph(RenderGraph renderGraph, ContextContainer frameData)
        {
            var resources=frameData.Get<UniversalResourceData>();
            if (!resources.activeColorTexture.IsValid()) return;
            using(var builder=renderGraph.AddRasterRenderPass<PassData>("NBFX T06 Mask View",out var data))
            {
                data.material=material;
                builder.UseGlobalTexture(MaskId,AccessFlags.Read);
                builder.SetRenderAttachment(resources.activeColorTexture,0,AccessFlags.ReadWrite);
                builder.AllowPassCulling(false);
                builder.SetRenderFunc((PassData d,RasterGraphContext context)=>
                { context.cmd.DrawProcedural(Matrix4x4.identity,d.material,0,MeshTopology.Triangles,3,1); });
            }
        }
    }
}
