#!/usr/bin/env python3
"""One-off Shader Graph 17.3 Dissolve Ramp serialization, input->output only."""
from pathlib import Path
import importlib.util,copy,sys,hashlib
spec=importlib.util.spec_from_file_location('base','/tmp/nbfx-fresnel-graph-generator.py')
b=importlib.util.module_from_spec(spec);spec.loader.exec_module(b)
b.PREFIX='nbfx/nbshadergraph/dissolveramp/v1/'
src,dst=map(Path,sys.argv[1:3]);assert src.resolve()!=dst.resolve() and not dst.exists()
raw=src.read_text();objects=b.parse_stream(raw);byid=b.index_objects(objects)
graph=b.exactly_one(objects,lambda x:x.get('m_Type')==b.GRAPH_TYPE,'GraphData')
cf=b.exactly_one(objects,lambda x:x.get('m_Type')==b.CF_TYPE and x.get('m_FunctionName')=='NBGraphBaseColor','CF')
category=byid[graph['m_CategoryData'][0]['m_Id']]
original_cf_slots=[byid[x['m_Id']] for x in cf['m_Slots']]
assert {byid[x['m_Id']]['m_Id'] for x in cf['m_Slots']}==set(range(103))
# name, exact ShaderLab reference, template reference, value, CF template slot
pdefs=[
 ('DissolveRampToggle','_Dissolve_useRampMap_Toggle','_RampColorToggle',0.0,75),
 ('DissolveRampMap','_DissolveRampMap','_RampColorMap',None,74),
 ('DissolveRampSourceMode','_DissolveRampSourceMode','_RampColorSourceMode',0.0,76),
 ('DissolveRampColor','_DissolveRampColor','_RampColorBlendColor',(1,1,1,1),89),
 ('DissolveRampCount','_DissolveRampCount','_RampColorCount',131074.0,88),
 ('DissolveRampColor0','_DissolveRampColor0','_RampColor0',(1,0,0,0),79),
 ('DissolveRampColor1','_DissolveRampColor1','_RampColor1',(0,0,0,1),80),
 ('DissolveRampColor2','_DissolveRampColor2','_RampColor2',(1,1,1,1),81),
 ('DissolveRampColor3','_DissolveRampColor3','_RampColor3',(1,1,1,1),82),
 ('DissolveRampColor4','_DissolveRampColor4','_RampColor4',(1,1,1,1),83),
 ('DissolveRampColor5','_DissolveRampColor5','_RampColor5',(1,1,1,1),84),
 ('DissolveRampAlpha0','_DissolveRampAlpha0','_RampColorAlpha0',(1,0,1,1),85),
 ('DissolveRampAlpha1','_DissolveRampAlpha1','_RampColorAlpha1',(1,0,1,1),86),
 ('DissolveRampAlpha2','_DissolveRampAlpha2','_RampColorAlpha2',(1,0,1,1),87),
 ('NB_ForceNoMipFlagsLo16','_NB_ForceNoMipFlagsLo16','_NB_WrapFlagsLo16',0.0,39),
 ('NB_ForceNoMipFlagsHi16','_NB_ForceNoMipFlagsHi16','_NB_WrapFlagsHi16',0.0,40),
]
for _,ref,_,_,_ in pdefs:assert not any(x.get('m_DefaultReferenceName')==ref for x in objects),ref
new=[]
for idx,(name,ref,template_ref,val,cf_idx) in enumerate(pdefs,103):
 tp=b.property_template(objects,template_ref)
 prop=copy.deepcopy(tp);prop['m_ObjectId']=b.stable_id('property/'+name)
 prop['m_Guid']['m_GuidSerialized']=b.stable_guid('property/'+name+'/guid')
 prop['m_Name']=name;prop['m_RefNameGeneratedByDisplayName']=name
 prop['m_DefaultReferenceName']=ref;prop['m_OverrideReferenceName']=ref
 if val is not None:
  if isinstance(val,tuple):
   prop['m_Value']=dict(zip(('r','g','b','a') if 'ColorShaderProperty' in tp['m_Type'] else ('x','y','z','w'),val))
  else:prop['m_Value']=val
 if ref=='_DissolveRampColor':prop['m_ColorMode']=1
 tnode=b.exactly_one(objects,lambda x:x.get('m_Type')==b.PROPERTY_NODE_TYPE and x.get('m_Property',{}).get('m_Id')==tp['m_ObjectId'],'node '+template_ref)
 n=copy.deepcopy(tnode);n['m_ObjectId']=b.stable_id('property/'+name+'/node');n['m_Property']['m_Id']=prop['m_ObjectId']
 oslot=copy.deepcopy(byid[tnode['m_Slots'][0]['m_Id']]);oslot['m_ObjectId']=b.stable_id('property/'+name+'/output');oslot['m_DisplayName']=name
 n['m_Slots']=[{'m_Id':oslot['m_ObjectId']}];b.position(n,620,3300+(idx-103)*85)
 cf_template=b.exactly_one(original_cf_slots,lambda x:x['m_Id']==cf_idx,'CF template '+str(cf_idx))
 inslot=copy.deepcopy(cf_template);inslot['m_ObjectId']=b.stable_id('cf/slot/'+str(idx)+'/'+name)
 inslot['m_Id']=idx;inslot['m_DisplayName']=name;inslot['m_ShaderOutputName']=name
 if val is not None:
  if isinstance(val,tuple):
   vec=dict(zip(('x','y','z','w'),val));inslot['m_Value']=vec.copy();inslot['m_DefaultValue']=vec.copy()
  else:inslot['m_Value']=val;inslot['m_DefaultValue']=val
 new.extend((prop,n,oslot,inslot));graph['m_Properties'].append({'m_Id':prop['m_ObjectId']});category['m_ChildObjectList'].append({'m_Id':prop['m_ObjectId']});graph['m_Nodes'].append({'m_Id':n['m_ObjectId']});cf['m_Slots'].append({'m_Id':inslot['m_ObjectId']});graph['m_Edges'].append(b.edge(n['m_ObjectId'],cf['m_ObjectId'],idx))
objects.extend(new);emitted=b.serialize_stream(objects);parsed=b.parse_stream(emitted);ids=b.index_objects(parsed)
assert len(ids)==len(parsed)
assert {ids[x['m_Id']]['m_Id'] for x in cf['m_Slots']}==set(range(119))
for _,ref,_,_,_ in pdefs:assert len([x for x in parsed if x.get('m_DefaultReferenceName')==ref])==1
for e in graph['m_Edges']:
 assert e['m_OutputSlot']['m_Node']['m_Id'] in ids and e['m_InputSlot']['m_Node']['m_Id'] in ids
assert not dst.exists();dst.write_text(emitted)
print('source_sha256',hashlib.sha256(raw.encode()).hexdigest(),'output_sha256',hashlib.sha256(emitted.encode()).hexdigest(),'objects',len(parsed))
