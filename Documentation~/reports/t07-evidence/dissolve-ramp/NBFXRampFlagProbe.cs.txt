using System;
using System.Reflection;
using UnityEditor;
using UnityEngine;
using NBShaderEditor;
public static class NBFXRampFlagProbe
{
    public static void Run()
    {
        var shader=AssetDatabase.LoadAssetAtPath<Shader>(
            "Packages/com.xuanxuan.nb.fx/NBShaders2/ShaderGraph/NBShaderGraph.shadergraph");
        if(!shader||!shader.isSupported)throw new Exception("Graph missing");
        var t=typeof(NBShaderGraphRootItem);
        var setFlag=t.GetMethod("SetPackedFlag",BindingFlags.Static|BindingFlags.NonPublic);
        var setWrap=t.GetMethod("SetPackedWrapMode",BindingFlags.Static|BindingFlags.NonPublic);
        if(setFlag==null||setWrap==null)throw new Exception("GUI bit helpers missing");
        var a=new Material(shader);var b=new Material(shader);
        try{
            a.SetFloat("_NB_Flags1Lo16",0xA001);b.SetFloat("_NB_Flags1Lo16",0x0101);
            a.SetFloat("_NB_ForceNoMipFlagsHi16",0xC000);
            a.SetFloat("_NB_WrapFlagsLo16",0xA001);a.SetFloat("_NB_WrapFlagsHi16",0x4001);
            bool Flag(Material m,string p,int bit,bool v)=>(bool)setFlag.Invoke(null,new object[]{m,p,bit,v});
            bool Wrap(Material m,int mode)=>(bool)setWrap.Invoke(null,new object[]{m,"_NB_WrapFlagsLo16","_NB_WrapFlagsHi16",128,mode});
            if(!Flag(a,"_NB_Flags1Lo16",64,true)||a.GetFloat("_NB_Flags1Lo16")!=0xA041)throw new Exception("bit6 A");
            if(!Flag(b,"_NB_Flags1Lo16",64,true)||b.GetFloat("_NB_Flags1Lo16")!=0x0141)throw new Exception("bit6 B");
            if(Flag(a,"_NB_Flags1Lo16",64,true))throw new Exception("bit6 duplicate");
            if(!Flag(a,"_NB_ForceNoMipFlagsHi16",1,true)||a.GetFloat("_NB_ForceNoMipFlagsHi16")!=0xC001)throw new Exception("force bit16");
            for(int mode=0;mode<4;mode++){
                if(!Wrap(a,mode)&&mode!=0)throw new Exception("wrap change missing "+mode);
                int lo=(int)a.GetFloat("_NB_WrapFlagsLo16");int hi=(int)a.GetFloat("_NB_WrapFlagsHi16");
                int actual=((lo&128)!=0?1:0)|((hi&128)!=0?2:0);
                if(actual!=mode||(lo&~128)!=(0xA001&~128)||(hi&~128)!=(0x4001&~128))
                    throw new Exception("wrap mode/preserve "+mode+" "+lo+" "+hi);
            }
            if(Wrap(a,3))throw new Exception("wrap duplicate");
            if(b.GetFloat("_NB_Flags1Lo16")!=0x0141)throw new Exception("B isolation");
            Debug.Log("NBFX_RAMP_FLAG_PROBE PASS bit6/forceNoMip16/wrap0..3 preserve unrelated bits");
        }
        finally{UnityEngine.Object.DestroyImmediate(a);UnityEngine.Object.DestroyImmediate(b);}
    }
}
