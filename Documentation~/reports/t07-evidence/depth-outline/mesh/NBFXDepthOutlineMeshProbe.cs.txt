using System;

    static Material MakeGraph(Shader shader)
    {var m=new Material(shader);m.SetTexture("_BaseMap",Texture2D.whiteTexture);m.SetColor("_Color",new Color(Mathf.LinearToGammaSpace(.2f),Mathf.LinearToGammaSpace(.3f),Mathf.LinearToGammaSpace(.4f),.6f));m.SetColor("_ColorA",Color.white);m.SetFloat("_NB_Flags0Lo16",0);m.SetFloat("_NB_Flags0Hi16",0);m.SetFloat("_NB_Flags1Lo16",0);m.SetFloat("_NB_Flags1Hi16",0);m.SetFloat("_NB_DistortionMode",0);m.SetFloat("_Cull",0);m.SetFloat("_Surface",1);m.SetFloat("_ZWrite",0);m.SetFloat("_ZTest",4);m.SetFloat("_SrcBlend",(float)BlendMode.One);m.SetFloat("_DstBlend",(float)BlendMode.Zero);m.SetFloat("_SrcBlendAlpha",(float)BlendMode.One);m.SetFloat("_DstBlendAlpha",(float)BlendMode.Zero);m.EnableKeyword("_SURFACE_TYPE_TRANSPARENT");m.renderQueue=3000;m.SetVector("_DepthOutline_Vec",new Vector4(0,1,0,0));m.SetColor("_DepthOutline_Color",new Color(1,.1f,.1f,1));return m;}

    try{foreach(bool ortho in new[]{true,false})foreach(bool hasBack in new[]{false,true})foreach(bool enabled in new[]{false,true})
    {cam.orthographic=ortho;back.SetActive(hasBack);graph.SetFloat("_DepthOutline_Toggle",enabled?1:0);legacy.SetFloat("_DepthOutline_Toggle",enabled?1:0);if(enabled)legacy.EnableKeyword("_DEPTH_OUTLINE");else legacy.DisableKeyword("_DEPTH_OUTLINE");string key=(ortho?"ortho":"persp")+"-"+(hasBack?"occluder":"empty")+"-"+(enabled?"on":"off");front.GetComponent<MeshRenderer>().sharedMaterial=graph;var a=Capture(cam,rt,rb,"graph-"+key);front.GetComponent<MeshRenderer>().sharedMaterial=legacy;var b=Capture(cam,rt,rb,"legacy-"+key);int diff=AlphaDiff(a,b);Debug.Log("NBFX_DEPTH_OUTLINE_MESH "+key+" graph="+a[48*96+48]+" legacy="+b[48*96+48]+" alphaDiff="+diff);Need(diff==0,"strict alpha mismatch "+key);if(enabled&&hasBack)Need(a[48*96+48].a>200&&a[48*96+48].g<50,"outline inactive "+key);if(!enabled||!hasBack)Need(a[48*96+48].a==153,"unexpected outline "+key);}
    Debug.Log("NBFX_DEPTH_OUTLINE_MESH DONE graphPass="+graph.passCount+" legacyPass="+legacy.passCount+" graphMessages="+ShaderUtil.GetShaderMessages(sg).Length+" legacyMessages="+ShaderUtil.GetShaderMessages(sl).Length);foreach(var msg in ShaderUtil.GetShaderMessages(sg))Debug.Log("NBFX_DEPTH_OUTLINE_SHADER_MESSAGE "+msg.severity+" "+msg.message);
    }finally{UnityEngine.Object.DestroyImmediate(front);UnityEngine.Object.DestroyImmediate(back);UnityEngine.Object.DestroyImmediate(holder);UnityEngine.Object.DestroyImmediate(graph);UnityEngine.Object.DestroyImmediate(legacy);UnityEngine.Object.DestroyImmediate(opaque);UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(rb);}}
}
