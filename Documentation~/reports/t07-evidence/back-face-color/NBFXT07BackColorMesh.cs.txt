using System;
using System.IO;
using UnityEditor;
using UnityEngine;
public static class NBFXT07BackColorMesh
{
 const string Graph="Packages/com.xuanxuan.nb.fx/NBShaders2/ShaderGraph/NBShaderGraph.shadergraph";
 const string Root="/tmp/nbfx-backcolor-20260930";
 static Color32[] Capture(Camera cam,RenderTexture rt,Texture2D rb,string name){cam.Render();var prev=RenderTexture.active;RenderTexture.active=rt;rb.ReadPixels(new Rect(0,0,96,96),0,0);rb.Apply(false);RenderTexture.active=prev;File.WriteAllBytes(Root+"/"+name+".png",rb.EncodeToPNG());return rb.GetPixels32();}
 static int Diff(Color32[] a,Color32[] b){int n=0;for(int i=0;i<a.Length;i++)if(Math.Abs(a[i].r-b[i].r)>2||Math.Abs(a[i].g-b[i].g)>2||Math.Abs(a[i].b-b[i].b)>2)n++;return n;}
 public static void Run(){Directory.CreateDirectory(Root);AssetDatabase.ImportAsset(Graph,ImportAssetOptions.ForceUpdate);var shader=AssetDatabase.LoadAssetAtPath<Shader>(Graph);if(!shader||!shader.isSupported)throw new Exception("shader missing");
  var quad=GameObject.CreatePrimitive(PrimitiveType.Quad);quad.layer=2;quad.transform.localScale=new Vector3(2,2,1);
  var mat=new Material(shader);mat.SetColor("_Color",Color.white);mat.SetColor("_BaseBackColor",Color.red);mat.SetFloat("_Cull",0);mat.SetFloat("_Surface",0);mat.SetFloat("_ZWrite",1);mat.SetFloat("_NB_DistortionMode",0);mat.SetFloat("_NB_Flags0Lo16",0);mat.SetFloat("_NB_Flags0Hi16",0);quad.GetComponent<MeshRenderer>().sharedMaterial=mat;
  var go=new GameObject("backcolor camera");var cam=go.AddComponent<Camera>();cam.orthographic=true;cam.orthographicSize=1.5f;cam.nearClipPlane=.1f;cam.farClipPlane=20;cam.cullingMask=1<<2;cam.transform.position=new Vector3(0,0,5);cam.transform.rotation=Quaternion.Euler(0,180,0);cam.clearFlags=CameraClearFlags.SolidColor;cam.backgroundColor=Color.black;cam.allowHDR=false;cam.allowMSAA=false;
  var rt=new RenderTexture(96,96,24,RenderTextureFormat.ARGB32);rt.Create();cam.targetTexture=rt;var rb=new Texture2D(96,96,TextureFormat.RGBA32,false,true);
  try{
   quad.transform.rotation=Quaternion.Euler(0,180,0);var frontOff=Capture(cam,rt,rb,"front-off");
   quad.transform.rotation=Quaternion.identity;var backOff=Capture(cam,rt,rb,"back-off");
   mat.SetFloat("_NB_Flags0Hi16",4096);
   quad.transform.rotation=Quaternion.Euler(0,180,0);var frontOn=Capture(cam,rt,rb,"front-on");
   quad.transform.rotation=Quaternion.identity;var backOn=Capture(cam,rt,rb,"back-on");
   int offFaceDiff=Diff(frontOff,backOff),frontFlagDiff=Diff(frontOff,frontOn),backFlagDiff=Diff(backOff,backOn);
   Color32 a=frontOn[48*96+48],b=backOn[48*96+48];
   Debug.Log("NBFX_BACK_MESH offFaceDiff="+offFaceDiff+" frontFlagDiff="+frontFlagDiff+" backFlagDiff="+backFlagDiff+" front="+a+" back="+b+" passes="+mat.passCount+" opaque="+mat.FindPass("NBCameraOpaqueDistortPass")+" deferred="+mat.FindPass("NBDeferredDistortPass")+" messages="+ShaderUtil.GetShaderMessages(shader).Length);
   if(offFaceDiff!=0||frontFlagDiff!=0||backFlagDiff<500||a.g<200||b.r<200||b.g>30||mat.FindPass("NBCameraOpaqueDistortPass")!=1||mat.FindPass("NBDeferredDistortPass")!=2)throw new Exception("BackColor controlled Mesh behavior failed");mat.SetFloat("_Cull",2);quad.transform.rotation=Quaternion.identity;var identityCull=Capture(cam,rt,rb,"identity-cullback");quad.transform.rotation=Quaternion.Euler(0,180,0);var rotatedCull=Capture(cam,rt,rb,"rotated-cullback");Debug.Log("NBFX_BACK_CULL identity="+identityCull[48*96+48]+" rotated="+rotatedCull[48*96+48]);
  }finally{UnityEngine.Object.DestroyImmediate(quad);UnityEngine.Object.DestroyImmediate(go);UnityEngine.Object.DestroyImmediate(mat);UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(rb);}
 }
}
