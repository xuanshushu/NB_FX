// /tmp-only controlled Graph Mesh / ShaderLab line-color B/C probe.
using System;
using System.Collections.Generic;
using System.IO;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;

public static class NBFXLineMeshProbe
{
    const string Root="/tmp/nbfx-vfx-dissolve-line-probe-20260930/run/mesh";
    const string GraphPath="Packages/com.xuanxuan.nb.fx/NBShaders2/ShaderGraph/NBShaderGraph.shadergraph";
    const string LegacyPath="Packages/com.xuanxuan.nb.fx/NBShaders2/Shader/NBShader.shader";
    const string LegacySample="Assets/NBShaderSamples/NBShaderSamples/UnLit.mat";
    const int Size=128;
    [Serializable] sealed class CaseInfo
    {
        public string name;
        public int visiblePixels;
        public float effectiveAlpha, sourceR, sourceG, sourceB;
    }
    [Serializable] sealed class Result
    {
        public string status,error,note;
        public bool graphFunctional,legacyFunctional,graphLegacyBC;
        public CaseInfo graphOff,graphOn,legacyOff,legacyOn;
    }
    sealed class Fixture
    {
        public GameObject quad,cameraObject;
        public Camera camera;
        public RenderTexture target;
        public Texture2D readback,gray,white;
        public Material graphOff,graphOn,legacyOff,legacyOn;
    }
    static Color[] Capture(Fixture f,Material material,Color clear,string file)
    {
        f.quad.GetComponent<MeshRenderer>().sharedMaterial=material;
        f.camera.backgroundColor=clear;
        for(int i=0;i<3;i++)f.camera.Render();
        var previous=RenderTexture.active;
        try{
            RenderTexture.active=f.target;
            f.readback.ReadPixels(new Rect(0,0,Size,Size),0,0);
            f.readback.Apply(false);
            var colors=f.readback.GetPixels();
            var png=new Texture2D(Size,Size,TextureFormat.RGBA32,false,true);
            try{png.SetPixels(colors);png.Apply(false);
                File.WriteAllBytes(Path.Combine(Root,file+".png"),png.EncodeToPNG());}
            finally{UnityEngine.Object.DestroyImmediate(png);}
            return colors;
        }finally{RenderTexture.active=previous;}
    }
    static float Diff(Color a,Color b)=>Mathf.Max(
        Mathf.Abs(a.r-b.r),Mathf.Abs(a.g-b.g),Mathf.Abs(a.b-b.b));
    static float Median(List<float> a)
    {
        if(a.Count==0)throw new Exception("No safe Mesh pixels");
        a.Sort();return a[a.Count/2];
    }
    static CaseInfo Describe(string name,Color[] b0,Color[] b1,
        Color[] f0,Color[] f1)
    {
        var alpha=new List<float>();var red=new List<float>();
        var green=new List<float>();var blue=new List<float>();
        int visible=0;
        for(int y=Size/4;y<Size*3/4;y++)
            for(int x=Size/4;x<Size*3/4;x++)
            {
                int i=y*Size+x;
                if(Diff(b0[i],f0[i])<.03f)continue;
                visible++;
                float a=0;
                for(int ch=0;ch<3;ch++)
                {
                    float denominator=b1[i][ch]-b0[i][ch];
                    if(denominator<.8f)
                        throw new Exception("Black/white baseline mismatch");
                    a+=1f-(f1[i][ch]-f0[i][ch])/denominator;
                }
                a/=3f;
                if(a<.01f||a>1.05f)
                    throw new Exception(name+" unphysical alpha "+a);
                alpha.Add(a);
                red.Add((f0[i].r-(1-a)*b0[i].r)/a);
                green.Add((f0[i].g-(1-a)*b0[i].g)/a);
                blue.Add((f0[i].b-(1-a)*b0[i].b)/a);
            }
        return new CaseInfo{name=name,visiblePixels=visible,
            effectiveAlpha=Median(alpha),sourceR=Median(red),
            sourceG=Median(green),sourceB=Median(blue)};
    }
    static Material GraphMaterial(Shader shader,Texture2D gray,Texture2D white,bool line)
    {
        var m=new Material(shader);
        m.SetTexture("_BaseMap",white);
        // SG Color property is converted to linear in the generated Graph;
        // compensate in the test material so both hosts receive the same RGB.
        Color graphTint=new Color(.25f,.5f,.625f,1).gamma;
        graphTint.a=.6666667f;m.SetColor("_Color",graphTint);
        m.SetTexture("_DissolveMap",gray);
        m.SetFloat("_Dissolve_Toggle",1);
        m.SetVector("_Dissolve",new Vector4(.5f,1,1,.1f));
        m.SetVector("_Dissolve_Vec2",new Vector4(.2f,.1f,0,0));
        m.SetColor("_DissolveLineColor",new Color(1,0,0,1));
        m.SetFloat("_NB_Flags1Lo16",line?32:0);
        m.SetFloat("_NB_DistortionMode",0);
        m.SetFloat("_Surface",1);m.SetFloat("_ZWrite",0);
        m.SetFloat("_SrcBlend",(float)BlendMode.SrcAlpha);
        m.SetFloat("_DstBlend",(float)BlendMode.OneMinusSrcAlpha);
        m.SetFloat("_SrcBlendAlpha",(float)BlendMode.One);
        m.SetFloat("_DstBlendAlpha",(float)BlendMode.OneMinusSrcAlpha);
        m.SetFloat("_Cull",0);m.renderQueue=3000;
        m.EnableKeyword("_SURFACE_TYPE_TRANSPARENT");
        return m;
    }
    static Material LegacyMaterial(Material sample,Shader shader,
        Texture2D gray,Texture2D white,bool line)
    {
        var m=new Material(sample);m.shader=shader;
        m.SetTexture("_BaseMap",white);m.SetTexture("_MainTex",white);
        m.SetColor("_BaseColor",new Color(.25f,.5f,.625f,.6666667f));
        m.SetTexture("_DissolveMap",gray);
        m.SetFloat("_Dissolve_Toggle",1);
        m.SetVector("_Dissolve",new Vector4(.5f,1,1,.1f));
        m.SetVector("_Dissolve_Vec2",new Vector4(.2f,.1f,0,0));
        m.SetColor("_DissolveLineColor",new Color(1,0,0,1));
        m.SetFloat("_DissolveLineMaskToggle",line?1:0);
        m.SetInteger("_W9ParticleShaderFlags",0);
        m.SetInteger("_W9ParticleShaderFlags1",line?33:1);
        m.SetInteger("_W9ParticleColorChannelFlag0",0);
        m.SetFloat("_SrcBlend",(float)BlendMode.SrcAlpha);
        m.SetFloat("_DstBlend",(float)BlendMode.OneMinusSrcAlpha);
        m.SetFloat("_ZWrite",0);m.SetFloat("_ZTest",4);
        m.SetFloat("_Cull",0);m.SetFloat("_ColorMask",15);
        m.renderQueue=3000;
        m.EnableKeyword("_FX_LIGHT_MODE_UNLIT");m.EnableKeyword("_DISSOLVE");
        m.DisableKeyword("_DISSOLVE_RAMP");m.DisableKeyword("_DISSOLVE_MASK");
        m.DisableKeyword("_EMISSION");
        return m;
    }
    static Fixture Setup()
    {
        AssetDatabase.ImportAsset(GraphPath,ImportAssetOptions.ForceUpdate);
        AssetDatabase.ImportAsset(LegacyPath,ImportAssetOptions.ForceUpdate);
        var graph=AssetDatabase.LoadAssetAtPath<Shader>(GraphPath);
        var legacy=AssetDatabase.LoadAssetAtPath<Shader>(LegacyPath);
        var sample=AssetDatabase.LoadAssetAtPath<Material>(LegacySample);
        if(!graph||!legacy||!sample||!graph.isSupported||!legacy.isSupported)
            throw new Exception("Graph, legacy shader, or legacy sample unavailable");
        if(!SystemInfo.SupportsRenderTextureFormat(RenderTextureFormat.ARGBHalf))
            throw new Exception("Requires ARGBHalf RT");
        var f=new Fixture();
        f.gray=new Texture2D(1,1,TextureFormat.RGBA32,false,true);
        f.gray.SetPixel(0,0,new Color(133f/255f,133f/255f,133f/255f,1));
        f.gray.Apply(false);f.gray.filterMode=FilterMode.Point;
        f.gray.wrapMode=TextureWrapMode.Clamp;
        f.white=new Texture2D(1,1,TextureFormat.RGBA32,false,true);
        f.white.SetPixel(0,0,Color.white);f.white.Apply(false);
        f.white.filterMode=FilterMode.Point;f.white.wrapMode=TextureWrapMode.Clamp;
        f.graphOff=GraphMaterial(graph,f.gray,f.white,false);
        f.graphOn=GraphMaterial(graph,f.gray,f.white,true);
        f.legacyOff=LegacyMaterial(sample,legacy,f.gray,f.white,false);
        f.legacyOn=LegacyMaterial(sample,legacy,f.gray,f.white,true);
        f.quad=GameObject.CreatePrimitive(PrimitiveType.Quad);
        f.quad.layer=2;f.quad.transform.rotation=Quaternion.Euler(0,180,0);
        f.quad.transform.localScale=new Vector3(2,2,1);
        f.cameraObject=new GameObject("NBFX line Mesh B/C camera");
        f.camera=f.cameraObject.AddComponent<Camera>();
        f.camera.orthographic=true;f.camera.orthographicSize=1.5f;
        f.camera.nearClipPlane=.1f;f.camera.farClipPlane=20;
        f.camera.cullingMask=1<<2;
        f.camera.transform.position=new Vector3(0,0,5);
        f.camera.transform.rotation=Quaternion.Euler(0,180,0);
        f.camera.clearFlags=CameraClearFlags.SolidColor;
        f.camera.backgroundColor=Color.black;
        f.camera.allowHDR=true;f.camera.allowMSAA=false;
        f.target=new RenderTexture(Size,Size,24,RenderTextureFormat.ARGBHalf,
            RenderTextureReadWrite.Linear);f.target.Create();
        if(!f.target.IsCreated()||f.target.sRGB)
            throw new Exception("Linear ARGBHalf target failed");
        f.camera.targetTexture=f.target;
        f.readback=new Texture2D(Size,Size,TextureFormat.RGBAHalf,false,true);
        return f;
    }
    static void Cleanup(Fixture f)
    {
        if(f==null)return;
        if(f.camera)f.camera.targetTexture=null;
        if(f.target){f.target.Release();UnityEngine.Object.DestroyImmediate(f.target);}
        foreach(var m in new[]{f.graphOff,f.graphOn,f.legacyOff,f.legacyOn})
            if(m)UnityEngine.Object.DestroyImmediate(m);
        foreach(var t in new[]{f.readback,f.gray,f.white})
            if(t)UnityEngine.Object.DestroyImmediate(t);
        if(f.quad)UnityEngine.Object.DestroyImmediate(f.quad);
        if(f.cameraObject)UnityEngine.Object.DestroyImmediate(f.cameraObject);
    }
    static bool Functional(CaseInfo off,CaseInfo on)
    {
        return off.visiblePixels>100&&on.visiblePixels>100&&
            off.effectiveAlpha>.15f&&off.effectiveAlpha<.9f&&
            Mathf.Abs(on.effectiveAlpha-off.effectiveAlpha)<.035f&&
            on.sourceR>off.sourceR+.25f&&
            on.sourceG<off.sourceG-.15f&&
            on.sourceB<off.sourceB-.15f;
    }
    static bool Close(CaseInfo a,CaseInfo b)
    {
        return Mathf.Abs(a.effectiveAlpha-b.effectiveAlpha)<.08f&&
            Mathf.Abs(a.sourceR-b.sourceR)<.1f&&
            Mathf.Abs(a.sourceG-b.sourceG)<.1f&&
            Mathf.Abs(a.sourceB-b.sourceB)<.1f;
    }
    public static void Run()
    {
        Directory.CreateDirectory(Root);
        var result=new Result {
            note="Single-quad linear black/white source-over inference; B/C means Graph vs ShaderLab under controlled raw material flags, not GUI or serialized production materials. Uses tolerance, not strict zero."
        };
        Fixture f=null;
        try {
            f=Setup();
            f.quad.SetActive(false);
            var black=Capture(f,f.graphOff,Color.black,"background-black");
            var white=Capture(f,f.graphOff,Color.white,"background-white");
            f.quad.SetActive(true);
            result.graphOff=Describe("GraphOff",black,white,
                Capture(f,f.graphOff,Color.black,"graph-off-black"),
                Capture(f,f.graphOff,Color.white,"graph-off-white"));
            result.graphOn=Describe("GraphOn",black,white,
                Capture(f,f.graphOn,Color.black,"graph-on-black"),
                Capture(f,f.graphOn,Color.white,"graph-on-white"));
            result.legacyOff=Describe("ShaderLabOff",black,white,
                Capture(f,f.legacyOff,Color.black,"shaderlab-off-black"),
                Capture(f,f.legacyOff,Color.white,"shaderlab-off-white"));
            result.legacyOn=Describe("ShaderLabOn",black,white,
                Capture(f,f.legacyOn,Color.black,"shaderlab-on-black"),
                Capture(f,f.legacyOn,Color.white,"shaderlab-on-white"));
            result.graphFunctional=Functional(result.graphOff,result.graphOn);
            result.legacyFunctional=Functional(result.legacyOff,result.legacyOn);
            result.graphLegacyBC=Close(result.graphOff,result.legacyOff)&&
                Close(result.graphOn,result.legacyOn);
            result.status=result.graphFunctional&&result.legacyFunctional&&
                result.graphLegacyBC?"PASS":"NUMERIC_MISMATCH";
        }
        catch(Exception ex){result.status="FIXTURE_INVALID";result.error=ex.ToString();}
        finally{Cleanup(f);}
        File.WriteAllText(Path.Combine(Root,"result.json"),JsonUtility.ToJson(result,true));
        Debug.Log("NBFX_LINE_MESH "+JsonUtility.ToJson(result));
        if(result.status!="PASS")throw new Exception("NBFX line Mesh "+result.status+
            "; inspect "+Root+"/result.json");
    }
}
