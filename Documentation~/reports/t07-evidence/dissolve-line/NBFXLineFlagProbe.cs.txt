using System;
using System.Reflection;
using UnityEditor;
using UnityEngine;
using NBShaderEditor;
public static class NBFXLineFlagProbe
{
    public static void Run()
    {
        var shader=AssetDatabase.LoadAssetAtPath<Shader>(
            "Packages/com.xuanxuan.nb.fx/NBShaders2/ShaderGraph/NBShaderGraph.shadergraph");
        if(!shader || !shader.isSupported)throw new Exception("Graph unavailable");
        var method=typeof(NBShaderGraphRootItem).GetMethod("SetPackedFlag",
            BindingFlags.Static|BindingFlags.NonPublic);
        if(method==null)throw new Exception("Packed flag writer missing");
        var a=new Material(shader);var b=new Material(shader);
        try {
            a.SetFloat("_NB_Flags1Lo16",0xA001);b.SetFloat("_NB_Flags1Lo16",0x0101);
            bool Call(Material m,bool value)=>(bool)method.Invoke(null,new object[]{m,"_NB_Flags1Lo16",32,value});
            if(!Call(a,true)||a.GetFloat("_NB_Flags1Lo16")!=0xA021)throw new Exception("A on/preserve");
            if(!Call(b,true)||b.GetFloat("_NB_Flags1Lo16")!=0x0121)throw new Exception("B on/preserve");
            if(Call(a,true))throw new Exception("A duplicate write");
            if(!Call(a,false)||a.GetFloat("_NB_Flags1Lo16")!=0xA001)throw new Exception("A off/preserve");
            if(b.GetFloat("_NB_Flags1Lo16")!=0x0121)throw new Exception("B independently preserved");
            Debug.Log("NBFX_LINE_FLAG_PROBE PASS word1 bit5 independent material writes");
        }
        finally{UnityEngine.Object.DestroyImmediate(a);UnityEngine.Object.DestroyImmediate(b);}
    }
}
