using System;
using System.IO;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
public static class NBFXT07Overlay2Mesh
{
 const string Root="/tmp/nbfx-overlay2-20260930";
 const string Graph="Packages/com.xuanxuan.nb.fx/NBShaders2/ShaderGraph/NBShaderGraph.shadergraph";
 static Color Capture(Camera cam,RenderTexture rt,Texture2D rb,string name){cam.Render();var prev=RenderTexture.active;RenderTexture.active=rt;rb.ReadPixels(new Rect(0,0,96,96),0,0);rb.Apply(false);RenderTexture.active=prev;var preview=new Texture2D(96,96,TextureFormat.RGBA32,false,true);preview.SetPixels(rb.GetPixels());preview.Apply(false);File.WriteAllBytes(Root+"/"+name+".png",preview.EncodeToPNG());UnityEngine.Object.DestroyImmediate(preview);return rb.GetPixel(48,48);}
 static void AssertNear(Color actual,Color expected,string label){float e=Mathf.Max(Mathf.Abs(actual.r-expected.r),Mathf.Abs(actual.g-expected.g),Mathf.Abs(actual.b-expected.b),Mathf.Abs(actual.a-expected.a));if(e>.025f)throw new Exception(label+" actual="+actual+" expected="+expected+" error="+e);}
 public static void Run(){Directory.CreateDirectory(Root);AssetDatabase.ImportAsset(Graph,ImportAssetOptions.ForceUpdate);var shader=AssetDatabase.LoadAssetAtPath<Shader>(Graph);if(!shader||!shader.isSupported)throw new Exception("Graph missing");
  var quad=GameObject.CreatePrimitive(PrimitiveType.Quad);quad.layer=2;quad.transform.rotation=Quaternion.Euler(0,180,0);quad.transform.localScale=new Vector3(2,2,1);
  var mat=new Material(shader);mat.SetColor("_Color",new Color(.2f,.3f,.4f,.6f));mat.SetColor("_ColorBlendColor",new Color(.5f,1f,.5f,.8f));mat.SetFloat("_ColorBlendColorIntensity",1.5f);mat.SetVector("_ColorBlendVec",new Vector4(0,0,.6f,0));mat.SetFloat("_ColorBlendMap_Toggle",0);mat.SetFloat("_NB_DistortionMode",0);mat.SetFloat("_NB_Flags0Lo16",0);mat.SetFloat("_NB_Flags0Hi16",0);mat.SetFloat("_NB_Flags1Lo16",0);mat.SetFloat("_NB_Flags1Hi16",0);mat.SetFloat("_Cull",2);mat.SetFloat("_Surface",1);mat.SetFloat("_ZWrite",0);mat.SetFloat("_SrcBlend",(float)BlendMode.One);mat.SetFloat("_DstBlend",(float)BlendMode.Zero);mat.SetFloat("_SrcBlendAlpha",(float)BlendMode.One);mat.SetFloat("_DstBlendAlpha",(float)BlendMode.Zero);mat.EnableKeyword("_SURFACE_TYPE_TRANSPARENT");mat.renderQueue=3000;
  var map=new Texture2D(1,1,TextureFormat.RGBA32,false,true);map.SetPixel(0,0,new Color(204f/255f,26f/255f,51f/255f,128f/255f));map.Apply(false);map.filterMode=FilterMode.Point;map.wrapMode=TextureWrapMode.Clamp;mat.SetTexture("_ColorBlendMap",map);quad.GetComponent<MeshRenderer>().sharedMaterial=mat;
  var go=new GameObject("overlay camera");var cam=go.AddComponent<Camera>();cam.orthographic=true;cam.orthographicSize=1.5f;cam.nearClipPlane=.1f;cam.farClipPlane=20;cam.cullingMask=1<<2;cam.transform.position=new Vector3(0,0,5);cam.transform.rotation=Quaternion.Euler(0,180,0);cam.clearFlags=CameraClearFlags.SolidColor;cam.backgroundColor=Color.black;cam.allowHDR=true;cam.allowMSAA=false;
  var rt=new RenderTexture(96,96,24,RenderTextureFormat.ARGBHalf,RenderTextureReadWrite.Linear);rt.Create();cam.targetTexture=rt;var rb=new Texture2D(96,96,TextureFormat.RGBAHalf,false,true);
  try{
   Color off=Capture(cam,rt,rb,"off");mat.SetFloat("_ColorBlendMap_Toggle",1);mat.SetFloat("_NB_Flags1Hi16",1);Color add=Capture(cam,rt,rb,"add");mat.SetFloat("_NB_Flags1Hi16",0);Color mul=Capture(cam,rt,rb,"multiply");mat.SetFloat("_NB_Flags1Hi16",1);mat.SetFloat("_NB_Flags0Hi16",512);Color addAlpha=Capture(cam,rt,rb,"add-alpha");mat.SetFloat("_NB_Flags1Hi16",0);Color mulAlpha=Capture(cam,rt,rb,"multiply-alpha");
   float a=128f/255f*.8f;a=Mathf.Lerp(1,a,.6f);
   var baseColor=new Color(Mathf.GammaToLinearSpace(.2f),Mathf.GammaToLinearSpace(.3f),Mathf.GammaToLinearSpace(.4f),.6f);var over=new Vector3(204f/255f,26f/255f,51f/255f);var tint=new Vector3(.75f,1.5f,.75f);var ot=Vector3.Scale(over,tint);
   var expectedAdd=new Color(baseColor.r+ot.x*a,baseColor.g+ot.y*a,baseColor.b+ot.z*a,baseColor.a);
   var expectedMul=new Color(Mathf.Lerp(baseColor.r,baseColor.r*ot.x,a),Mathf.Lerp(baseColor.g,baseColor.g*ot.y,a),Mathf.Lerp(baseColor.b,baseColor.b*ot.z,a),baseColor.a);
   var expectedAddAlpha=new Color(baseColor.r+ot.x,baseColor.g+ot.y,baseColor.b+ot.z,baseColor.a*a);
   var expectedMulAlpha=new Color(baseColor.r*ot.x,baseColor.g*ot.y,baseColor.b*ot.z,baseColor.a*a);
   Debug.Log("NBFX_OVERLAY2_MESH off="+off+" add="+add+" mul="+mul+" addAlpha="+addAlpha+" mulAlpha="+mulAlpha+" expectedAdd="+expectedAdd+" expectedMul="+expectedMul+" expectedAddAlpha="+expectedAddAlpha+" expectedMulAlpha="+expectedMulAlpha+" passes="+mat.passCount+" p1="+mat.FindPass("NBCameraOpaqueDistortPass")+" p2="+mat.FindPass("NBDeferredDistortPass")+" messages="+ShaderUtil.GetShaderMessages(shader).Length);
   AssertNear(off,baseColor,"off");AssertNear(add,expectedAdd,"add");AssertNear(mul,expectedMul,"multiply");AssertNear(addAlpha,expectedAddAlpha,"addAlpha");AssertNear(mulAlpha,expectedMulAlpha,"mulAlpha");if(mat.FindPass("NBCameraOpaqueDistortPass")!=1||mat.FindPass("NBDeferredDistortPass")!=2)throw new Exception("Pass regression");
  }finally{UnityEngine.Object.DestroyImmediate(quad);UnityEngine.Object.DestroyImmediate(go);UnityEngine.Object.DestroyImmediate(mat);UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(rb);UnityEngine.Object.DestroyImmediate(map);}
 }
}
