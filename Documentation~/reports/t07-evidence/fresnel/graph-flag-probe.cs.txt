using System;
using System.Reflection;
using UnityEditor;
using UnityEngine;
using NBShaderEditor;
public static class NBFXFresnelGraphFlagProbe
{
    public static void Run()
    {
        const string graphPath="Packages/com.xuanxuan.nb.fx/NBShaders2/ShaderGraph/NBShaderGraph.shadergraph";
        var shader=AssetDatabase.LoadAssetAtPath<Shader>(graphPath);
        if (!shader || !shader.isSupported) throw new Exception("Graph missing");
        var method=typeof(NBShaderGraphRootItem).GetMethod("SetPackedFlag",BindingFlags.Static|BindingFlags.NonPublic);
        if(method==null)throw new Exception("flag helper missing");
        var a=new Material(shader);var b=new Material(shader);
        try
        {
            a.SetFloat("_NB_Flags0Lo16",0xA001);b.SetFloat("_NB_Flags0Lo16",0x0101);
            a.SetFloat("_NB_Flags0Hi16",0xC001);b.SetFloat("_NB_Flags0Hi16",0x4001);
            bool Call(Material m,string prop,int bit,bool value)=>(bool)method.Invoke(null,new object[]{m,prop,bit,value});
            if(!Call(a,"_NB_Flags0Lo16",4,true)||a.GetFloat("_NB_Flags0Lo16")!=0xA005)throw new Exception("bit2 A");
            if(!Call(b,"_NB_Flags0Lo16",4,true)||b.GetFloat("_NB_Flags0Lo16")!=0x0105)throw new Exception("bit2 B");
            if(Call(a,"_NB_Flags0Lo16",4,true))throw new Exception("duplicate write");
            if(!Call(a,"_NB_Flags0Lo16",8192,false)||a.GetFloat("_NB_Flags0Lo16")!=0x8005)throw new Exception("bit13 preserve");
            if(!Call(a,"_NB_Flags0Hi16",4,true)||a.GetFloat("_NB_Flags0Hi16")!=0xC005)throw new Exception("bit18 A");
            if(!Call(b,"_NB_Flags0Hi16",4,true)||b.GetFloat("_NB_Flags0Hi16")!=0x4005)throw new Exception("bit18 B");
            if(!Call(a,"_NB_Flags0Hi16",4,false)||a.GetFloat("_NB_Flags0Hi16")!=0xC001)throw new Exception("bit18 off preserve");
            Debug.Log("NBFX_FRESNEL_GRAPH_FLAGS PASS independent multi-material slices and no-op");
        }
        finally{UnityEngine.Object.DestroyImmediate(a);UnityEngine.Object.DestroyImmediate(b);}
    }
}
