using System;
using System.IO;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
public static class NBFXT07RampMesh
{
 const string Graph="Packages/com.xuanxuan.nb.fx/NBShaders2/ShaderGraph/NBShaderGraph.shadergraph";
 const string Root="/tmp/nbfx-ramp-20260930";
 static Color[] Capture(Camera cam,RenderTexture rt,Texture2D rb,string name)
 {
  cam.Render();var prev=RenderTexture.active;RenderTexture.active=rt;rb.ReadPixels(new Rect(0,0,96,96),0,0);rb.Apply(false);RenderTexture.active=prev;
  var p=new Texture2D(96,96,TextureFormat.RGBA32,false,true);p.SetPixels(rb.GetPixels());p.Apply(false);File.WriteAllBytes(Root+"/"+name+".png",p.EncodeToPNG());UnityEngine.Object.DestroyImmediate(p);
  return new[]{rb.GetPixel(48,48),rb.GetPixel(32,48),rb.GetPixel(64,48)};
 }
 static void Near(float a,float b,float tol,string name){if(Mathf.Abs(a-b)>tol)throw new Exception(name+" actual="+a+" expected="+b);}
 public static void Run()
 {
  Directory.CreateDirectory(Root);AssetDatabase.ImportAsset(Graph,ImportAssetOptions.ForceUpdate);var shader=AssetDatabase.LoadAssetAtPath<Shader>(Graph);if(!shader||!shader.isSupported)throw new Exception("Graph missing");
  var quad=GameObject.CreatePrimitive(PrimitiveType.Quad);quad.layer=2;quad.transform.rotation=Quaternion.Euler(0,180,0);quad.transform.localScale=new Vector3(2,2,1);
  var mat=new Material(shader);mat.SetColor("_Color",new Color(.2f,.3f,.4f,.6f));mat.SetFloat("_RampColorToggle",0);mat.SetFloat("_RampColorSourceMode",0);mat.SetFloat("_RampColorCount",131074);mat.SetColor("_RampColor0",new Color(1,0,0,0));mat.SetColor("_RampColor1",new Color(0,1,0,1));mat.SetVector("_RampColorAlpha0",new Vector4(1,0,1,1));mat.SetColor("_RampColorBlendColor",new Color(1,1,1,.2f));mat.SetVector("_RampColorMapOffset",Vector4.zero);
  mat.SetFloat("_NB_ColorChannelLo16",3);mat.SetFloat("_NB_WrapFlagsLo16",0);mat.SetFloat("_NB_WrapFlagsHi16",0);mat.SetFloat("_NB_Flags0Lo16",0);mat.SetFloat("_NB_Flags0Hi16",0);mat.SetFloat("_NB_Flags1Lo16",0);mat.SetFloat("_NB_Flags1Hi16",0);mat.SetFloat("_NB_DistortionMode",0);mat.SetFloat("_Cull",2);mat.SetFloat("_Surface",1);mat.SetFloat("_ZWrite",0);mat.SetFloat("_SrcBlend",(float)BlendMode.One);mat.SetFloat("_DstBlend",(float)BlendMode.Zero);mat.SetFloat("_SrcBlendAlpha",(float)BlendMode.One);mat.SetFloat("_DstBlendAlpha",(float)BlendMode.Zero);mat.EnableKeyword("_SURFACE_TYPE_TRANSPARENT");mat.renderQueue=3000;
  var map=new Texture2D(1,1,TextureFormat.RGBA32,false,true);map.SetPixel(0,0,new Color(.25f,.75f,.5f,1));map.Apply(false);map.filterMode=FilterMode.Point;map.wrapMode=TextureWrapMode.Clamp;mat.SetTexture("_RampColorMap",map);quad.GetComponent<MeshRenderer>().sharedMaterial=mat;
  var go=new GameObject("ramp camera");var cam=go.AddComponent<Camera>();cam.orthographic=true;cam.orthographicSize=1.5f;cam.nearClipPlane=.1f;cam.farClipPlane=20;cam.cullingMask=1<<2;cam.transform.position=new Vector3(0,0,5);cam.transform.rotation=Quaternion.Euler(0,180,0);cam.clearFlags=CameraClearFlags.SolidColor;cam.backgroundColor=Color.black;cam.allowHDR=true;cam.allowMSAA=false;
  var rt=new RenderTexture(96,96,24,RenderTextureFormat.ARGBHalf,RenderTextureReadWrite.Linear);rt.Create();cam.targetTexture=rt;var rb=new Texture2D(96,96,TextureFormat.RGBAHalf,false,true);
  try
  {
   var off=Capture(cam,rt,rb,"off");mat.SetFloat("_RampColorToggle",1);var mul=Capture(cam,rt,rb,"multiply");mat.SetFloat("_NB_Flags0Hi16",256);var add=Capture(cam,rt,rb,"add");mat.SetFloat("_RampColorSourceMode",1);var tex=Capture(cam,rt,rb,"texture-source");
   Debug.Log("NBFX_RAMP_MESH off="+off[0]+" mul="+mul[0]+" add="+add[0]+" tex="+tex[0]+" left="+add[1]+" right="+add[2]+" passes="+mat.passCount+" p1="+mat.FindPass("NBCameraOpaqueDistortPass")+" p2="+mat.FindPass("NBDeferredDistortPass")+" messages="+ShaderUtil.GetShaderMessages(shader).Length);
   Near(off[0].r,Mathf.GammaToLinearSpace(.2f),.03f,"off.r");Near(off[0].a,.6f,.03f,"off.a");
   if(!(mul[0].b<.01f && mul[0].a<.2f && mul[0].r>0 && mul[0].g>0))throw new Exception("multiply ramp inactive");
   if(!(add[0].r>off[0].r+.2f && add[0].g>off[0].g+.2f && add[0].a>.75f && add[0].a<.85f))throw new Exception("add ramp inactive");
   if(!(add[1].r>add[2].r+.15f && add[1].g<add[2].g-.15f))throw new Exception("packed Ramp gradient/UV inactive");
   if(!(tex[0].r>add[0].r+.15f && tex[0].g<add[0].g-.15f))throw new Exception("Ramp texture source inactive");
   if(mat.FindPass("NBCameraOpaqueDistortPass")!=1||mat.FindPass("NBDeferredDistortPass")!=2)throw new Exception("custom Pass regression");
  }
  finally {UnityEngine.Object.DestroyImmediate(quad);UnityEngine.Object.DestroyImmediate(go);UnityEngine.Object.DestroyImmediate(mat);UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(rb);UnityEngine.Object.DestroyImmediate(map);}
 }
}
