// Test-only isolated clone Editor import/inspection/Player build.
using System;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text.RegularExpressions;
using UnityEditor;
using UnityEditor.Build.Reporting;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.VFX;

public static class NBFXRampPlayerBuild
{
    const string Root="/tmp/nbfx-vfx-ramp-probe-20260930/run";
    const string ScenePath="Assets/NBFXRampPlayer.unity";
    const string AppPath=Root+"/player/NBFXRampPlayer.app";
    const string GraphPath="Packages/com.xuanxuan.nb.fx/NBShaders2/ShaderGraph/NBShaderGraph.shadergraph";
    const string BaseTexturePath="Assets/NBFXRampBaseRGBA_64_128_160_192.png";
    const string BaseTextureGuid="394e89e2390e5c0897b40495ed873f98";
    const string RampTexturePath="Assets/NBFXRampSampleRGBA_64_200_50_128.png";
    const string RampTextureGuid="170d9609cc555711b0e665b932bc00e5";
    const long OutputId=8926484042661614853, BaseMapId=8926484042661614854;
    const long Flags0HiId=8926484042661614870;
    const long MapId=8926484042661616400, ToggleId=8926484042661616401;
    const long ModeId=8926484042661616402, CountId=8926484042661616403;
    const long OffsetId=8926484042661616410, TintId=8926484042661616510;
    static readonly long[] ColorIds={
        8926484042661616420,8926484042661616430,8926484042661616440,
        8926484042661616450,8926484042661616460,8926484042661616470};
    static readonly long[] AlphaIds={
        8926484042661616480,8926484042661616490,8926484042661616500};
    static readonly float[][] ColorValues={
        new[]{.2f,.8f,.3f,0f},new[]{.8f,.2f,.6f,1f},
        new[]{0f,0f,0f,0f},new[]{0f,0f,0f,0f},
        new[]{0f,0f,0f,0f},new[]{0f,0f,0f,0f}};
    static readonly float[][] AlphaValues={
        new[]{1f,0f,1f,1f},new[]{0f,0f,0f,0f},new[]{0f,0f,0f,0f}};
    static readonly string[] Names={"Off","Multiply","Add","UVMultiply"};
    static readonly string[] Guids={
        "aa1b5d393c0359eb92c6871e19eabfff","0b4adb508118556a8f0374df387a9efe",
        "51ac64b8f96a5f8eba912d5b97e618af","8f0a5c5d72ba50779ea2d06ace3848da"};
    static readonly int[] Toggle={0,1,1,1};
    static readonly int[] SourceMode={1,1,1,0};
    static readonly int[] Flags0Hi={0,0,256,0};

    public static void Run()
    {
        Directory.CreateDirectory(Root);
        using(var report=new StreamWriter(Path.Combine(Root,"editor-inspection.txt")))
        {
            ValidateTexture(BaseTexturePath,BaseTextureGuid,
                new Color(64f/255f,128f/255f,160f/255f,192f/255f),report);
            ValidateTexture(RampTexturePath,RampTextureGuid,
                new Color(64f/255f,200f/255f,50f/255f,128f/255f),report);
            var graph=AssetDatabase.LoadAssetAtPath<Shader>(GraphPath);
            if(!graph || !graph.isSupported)throw new Exception("Product Graph unavailable");
            var assets=new VisualEffectAsset[Names.Length];var shaders=new Shader[Names.Length];
            for(int i=0;i<Names.Length;i++)
            {
                string path="Assets/NBFXRamp"+Names[i]+".vfx";
                AssetDatabase.ImportAsset(path,ImportAssetOptions.ForceUpdate);
                if(AssetDatabase.AssetPathToGUID(path)!=Guids[i])
                    throw new Exception("VFX GUID mismatch: "+path);
                ValidateSerialized(path,Toggle[i],SourceMode[i],Flags0Hi[i],report);
                assets[i]=AssetDatabase.LoadAssetAtPath<VisualEffectAsset>(path);
                shaders[i]=FindShader(path);
                if(!assets[i] || !shaders[i] || !shaders[i].isSupported ||
                    !shaders[i].name.Contains("Mesh"))
                    throw new Exception("Real VFX Mesh Output import failed: "+path);
                ValidateGenerated(path,Toggle[i],SourceMode[i],Flags0Hi[i],report);
                var material=new Material(shaders[i]);
                try{report.WriteLine(path+" lazyEditorPassCount="+material.passCount);}
                finally{UnityEngine.Object.DestroyImmediate(material);}
            }
            var scene=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);
            var host=new GameObject("NBFX Ramp VFX Player Probe");
            var probe=host.AddComponent<NBFXRampPlayerProbe>();
            probe.graphShader=graph;
            probe.offAsset=assets[0];probe.multiplyAsset=assets[1];
            probe.addAsset=assets[2];probe.uvMultiplyAsset=assets[3];
            probe.offShader=shaders[0];probe.multiplyShader=shaders[1];
            probe.addShader=shaders[2];probe.uvMultiplyShader=shaders[3];
            if(!EditorSceneManager.SaveScene(scene,ScenePath))
                throw new Exception("Could not save clone-only Player scene");
            report.Flush();
        }
        Directory.CreateDirectory(Path.GetDirectoryName(AppPath));
        var build=BuildPipeline.BuildPlayer(new BuildPlayerOptions{
            scenes=new[]{ScenePath},locationPathName=AppPath,
            target=BuildTarget.StandaloneOSX,options=BuildOptions.None});
        Debug.Log("NBFX_RAMP_BUILD result="+build.summary.result+
            " errors="+build.summary.totalErrors+" warnings="+build.summary.totalWarnings);
        if(build.summary.result!=BuildResult.Succeeded || build.summary.totalErrors!=0)
            throw new Exception("Ramp VFX Player build failed");
    }
    static void ValidateTexture(string path,string guid,Color expected,TextWriter report)
    {
        AssetDatabase.ImportAsset(path,ImportAssetOptions.ForceUpdate);
        var importer=AssetImporter.GetAtPath(path) as TextureImporter;
        var texture=AssetDatabase.LoadAssetAtPath<Texture2D>(path);
        if(!importer || !texture || AssetDatabase.AssetPathToGUID(path)!=guid ||
           importer.sRGBTexture || importer.mipmapEnabled || !importer.isReadable ||
           importer.textureCompression!=TextureImporterCompression.Uncompressed ||
           texture.width!=1 || texture.height!=1 ||
           importer.filterMode!=FilterMode.Point || importer.wrapMode!=TextureWrapMode.Clamp)
            throw new Exception("Explicit RGBA import contract failed: "+path);
        var pixel=texture.GetPixel(0,0);
        for(int i=0;i<4;i++)if(Mathf.Abs(pixel[i]-expected[i])>.003f)
            throw new Exception("RGBA pixel mismatch: "+path+" "+pixel);
        report.WriteLine(path+" linearRGBA="+pixel+" mip=false uncompressed point/clamp");
    }
    static string Body(string text,long id)
    {
        var m=Regex.Match(text,@"(?ms)^--- !u!114 &"+id+@"\n.*?(?=^--- !u!|\z)");
        if(!m.Success)throw new Exception("Missing VFX object "+id);
        return m.Value;
    }
    static bool Scalar(string text,long id,string name,int value)
    {
        string b=Body(text,id);
        return b.Contains("name: "+name)&&b.Contains("m_SerializableObject: "+value+"\n");
    }
    static bool SerializedVector(string text,long id,string name,string json)
    {
        string b=Body(text,id);
        return b.Contains("name: "+name)&&b.Contains(json);
    }
    static void ValidateSerialized(string path,int toggle,int mode,int flags0,TextWriter report)
    {
        string text=File.ReadAllText(Path.Combine(Directory.GetParent(Application.dataPath).FullName,path));
        string output=Body(text,OutputId);
        if(!text.Contains("  capacity: 1\n") || !output.Contains("  cullMode: 0\n") ||
           !output.Contains("  blendMode: 1\n") ||
           !Body(text,BaseMapId).Contains(BaseTextureGuid) ||
           !Body(text,MapId).Contains("name: _RampColorMap") ||
           !Body(text,MapId).Contains(RampTextureGuid) ||
           !Scalar(text,ToggleId,"_RampColorToggle",toggle) ||
           !Scalar(text,ModeId,"_RampColorSourceMode",mode) ||
           !Scalar(text,CountId,"_RampColorCount",131074) ||
           !Scalar(text,Flags0HiId,"_NB_Flags0Hi16",flags0) ||
           !SerializedVector(text,OffsetId,"_RampColorMapOffset",
               "{\"x\":0.0,\"y\":0.0,\"z\":0.0,\"w\":0.0}") ||
           !SerializedVector(text,TintId,"_RampColorBlendColor",
               "{\"r\":0.5,\"g\":0.5,\"b\":0.5,\"a\":0.2}"))
            throw new Exception("Ramp serialized Output contract invalid: "+path);
        for(int i=0;i<ColorIds.Length;i++)
            if(!SerializedVector(text,ColorIds[i],"_RampColor"+i,ColorJson(ColorValues[i])))
                throw new Exception("Serialized RampColor"+i+" invalid: "+path);
        for(int i=0;i<AlphaIds.Length;i++)
            if(!SerializedVector(text,AlphaIds[i],"_RampColorAlpha"+i,VectorJson(AlphaValues[i])))
                throw new Exception("Serialized RampColorAlpha"+i+" invalid: "+path);
        foreach(long id in new[]{MapId,ToggleId,ModeId,OffsetId,CountId,TintId})
            if(!output.Contains("  - {fileID: "+id+"}\n"))
                throw new Exception("Output missing Ramp slot "+id+": "+path);
        foreach(long id in ColorIds)if(!output.Contains("  - {fileID: "+id+"}\n"))
            throw new Exception("Output missing Ramp color slot "+id);
        foreach(long id in AlphaIds)if(!output.Contains("  - {fileID: "+id+"}\n"))
            throw new Exception("Output missing Ramp alpha slot "+id);
        report.WriteLine(path+" toggle="+toggle+" sourceMode="+mode+
            " flags0Hi="+flags0+" count=131074 sampleRGBA=(64,200,50,128)");
    }
    static string Num(float v)=>v.ToString("0.0",CultureInfo.InvariantCulture);
    static string ColorJson(float[] v)=>"{\"r\":"+Num(v[0])+",\"g\":"+Num(v[1])+
        ",\"b\":"+Num(v[2])+",\"a\":"+Num(v[3])+"}";
    static string VectorJson(float[] v)=>"{\"x\":"+Num(v[0])+",\"y\":"+Num(v[1])+
        ",\"z\":"+Num(v[2])+",\"w\":"+Num(v[3])+"}";
    static Shader FindShader(string path)
    {
        foreach(var obj in AssetDatabase.LoadAllAssetsAtPath(path))
            if(obj is Shader shader && shader.name.Contains("Mesh"))return shader;
        return null;
    }
    static bool SourceVector(string source,string name,float[] expected)
    {
        var m=Regex.Match(source,@"output\."+Regex.Escape(name)+@"\s*=\s*float4\(([^)]*)\)");
        if(!m.Success)return false;
        var parts=m.Groups[1].Value.Split(',');if(parts.Length!=4)return false;
        for(int i=0;i<4;i++)
        {
            float actual;
            if(!float.TryParse(parts[i].Trim(),NumberStyles.Float,
                CultureInfo.InvariantCulture,out actual) || Mathf.Abs(actual-expected[i])>.005f)
                return false;
        }
        return true;
    }
    static void ValidateGenerated(string path,int toggle,int mode,int flags0,TextWriter report)
    {
        const BindingFlags flags=BindingFlags.Public|BindingFlags.NonPublic|
            BindingFlags.Static|BindingFlags.Instance;
        Type type=null;
        foreach(var assembly in AppDomain.CurrentDomain.GetAssemblies())
        {type=assembly.GetType("UnityEditor.VFX.VisualEffectResource");if(type!=null)break;}
        if(type==null)throw new Exception("VFX generated-resource reflection unavailable");
        var resource=type.GetMethod("GetResourceAtPath",flags).Invoke(null,new object[]{path});
        if(resource==null)throw new Exception("VFX generated-resource missing: "+path);
        int count=(int)type.GetMethod("GetShaderSourceCount",flags).Invoke(resource,null);
        var getter=type.GetMethod("GetShaderSource",flags);
        bool found=false;
        for(int i=0;i<count;i++)
        {
            string source=(string)getter.Invoke(resource,new object[]{i});
            if(!source.Contains("output._RampColorToggle"))continue;
            File.WriteAllText(Path.Combine(Root,Path.GetFileNameWithoutExtension(path)+"-"+i+".shader"),source);
            bool values=source.Contains("output._RampColorToggle = (float)"+toggle+";") &&
                source.Contains("output._RampColorSourceMode = (float)"+mode+";") &&
                source.Contains("output._NB_Flags0Hi16 = (float)"+flags0+";") &&
                source.Contains("output._RampColorCount = (float)131074;") &&
                SourceVector(source,"_RampColorMapOffset",new[]{0f,0f,0f,0f}) &&
                SourceVector(source,"_RampColorBlendColor",new[]{.5f,.5f,.5f,.2f});
            for(int c=0;c<ColorIds.Length;c++)values&=SourceVector(source,"_RampColor"+c,ColorValues[c]);
            for(int a=0;a<AlphaIds.Length;a++)values&=SourceVector(source,"_RampColorAlpha"+a,AlphaValues[a]);
            bool passes=source.Contains("\"LightMode\" = \"NBCameraOpaqueDistortPass\"") &&
                source.Contains("\"LightMode\" = \"NBDeferredDistortPass\"");
            // The custom function may remain in an external include, rather
            // than being textually expanded into the generated shader.
            bool ramp=source.Contains("_RampColorMap")&&source.Contains("NBGraphBaseColor");
            report.WriteLine(path+" generatedSource="+i+" allValues="+values+
                " exactPasses="+passes+" rampMapAndFunction="+ramp);
            found|=values&&passes&&ramp;
        }
        if(!found)throw new Exception("Ramp Output bindings or custom Passes absent: "+path);
    }
}
