// Isolated test fixture. Real one-particle VFX Mesh Fresnel numeric Player test.
using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.VFX;

public sealed class NBFXFresnelPlayerProbe : MonoBehaviour
{
    public VisualEffectAsset offAsset, colorAsset, alphaModeAsset, invertAsset,
        affectAlphaAsset, rotationAsset;
    public Shader graphShader, offShader, colorShader, alphaModeShader,
        invertShader, affectAlphaShader, rotationShader;
    const int Size = 256;
    const float RepeatTolerance = 0.005f;
    static readonly string[] Names = {
        "Off", "Color", "AlphaMode", "Invert", "AffectAlpha", "Rotation"
    };
    static readonly Vector3 BaseRGB = new Vector3(64f / 255f, 128f / 255f, 160f / 255f);
    const float BaseAlpha = (192f / 255f) * 0.6666667f;

    [Serializable] sealed class CaseInfo
    {
        public string name;
        public int alive, passCount, opaquePass, deferredPass, visiblePixels,
            changedVsOffPixels;
        public bool culled;
        public float repeatMaxRGB, effectiveAlphaMedian, effectiveAlphaP10,
            effectiveAlphaP90, sourceRMedian, sourceGMedian, sourceBMedian;
    }
    [Serializable] sealed class Result
    {
        public string unityVersion, graphicsDevice, status, error, note;
        public bool linearTarget, fixtureValid, pass;
        public int roiPixels;
        public float blackBaselineMaxRGB, whiteBaselineMinRGB;
        public CaseInfo off, color, alphaMode, invert, affectAlpha, rotation;
    }
    sealed class Context
    {
        public Camera camera;
        public GameObject cameraObject;
        public RenderTexture target;
        public Texture2D readback;
        public VisualEffect[] effects;
    }
    static string OutputPath()
    {
        foreach (string arg in Environment.GetCommandLineArgs())
            if (arg.StartsWith("--nbfx-fresnel-output=", StringComparison.Ordinal))
                return arg.Substring("--nbfx-fresnel-output=".Length);
        return null;
    }

    IEnumerator Start()
    {
        string output = OutputPath();
        if (string.IsNullOrEmpty(output)) { Application.Quit(2); yield break; }
        Directory.CreateDirectory(output);
        var result = new Result {
            unityVersion = Application.unityVersion,
            graphicsDevice = SystemInfo.graphicsDeviceType.ToString(),
            note = "Real one-particle Cube/CullOff VFX Mesh Output; each paused effect is isolated by moving its GameObject layer between 0 and 31. Linear black/white clear pairs infer effective source-over alpha and source RGB. Two visible cube faces are expected, but not assumed for non-Off comparisons. All images and repeats are saved. No direct render-target-alpha claim."
        };
        Context context = null;
        int exitCode = 2;
        try { context = Setup(); }
        catch (Exception ex) { result.status = "FIXTURE_INVALID"; result.error = ex.ToString(); }
        if (context != null && result.error == null)
        {
            // Reuse the previously Player-validated one-particle Mesh schedule.
            for (int frame = 0; frame < 3; frame++) { context.camera.Render(); yield return null; }
            foreach (var effect in context.effects) effect.Simulate(1f / 60f, 21u);
            for (int frame = 0; frame < 2; frame++) { context.camera.Render(); yield return null; }
            foreach (var effect in context.effects) effect.pause = true;
            try
            {
                Run(context, output, result);
                result.fixtureValid = FixtureValid(result);
                result.pass = result.fixtureValid && NumericPass(result);
                result.status = result.pass ? "PASS" :
                    result.fixtureValid ? "NUMERIC_MISMATCH" : "FIXTURE_INVALID";
                exitCode = result.pass ? 0 : 1;
            }
            catch (Exception ex) { result.status = "FIXTURE_INVALID"; result.error = ex.ToString(); }
        }
        try { Cleanup(context); }
        catch (Exception ex) {
            result.status = "CLEANUP_FAILED";
            result.error = (result.error ?? "") + "\n" + ex;
            exitCode = 2;
        }
        File.WriteAllText(Path.Combine(output, "result.json"), JsonUtility.ToJson(result, true));
        Debug.Log("NBFX_FRESNEL_PLAYER " + JsonUtility.ToJson(result));
        Application.Quit(exitCode);
    }

    Context Setup()
    {
        var assets = new[] { offAsset, colorAsset, alphaModeAsset, invertAsset,
            affectAlphaAsset, rotationAsset };
        var shaders = new[] { offShader, colorShader, alphaModeShader, invertShader,
            affectAlphaShader, rotationShader };
        if (!graphShader) throw new Exception("Graph reference missing");
        foreach (var asset in assets) if (!asset) throw new Exception("VFX Output reference missing");
        foreach (var shader in shaders) if (!shader) throw new Exception("Generated VFX shader missing");
        if (SystemInfo.graphicsDeviceType != GraphicsDeviceType.Metal ||
            !SystemInfo.SupportsRenderTextureFormat(RenderTextureFormat.ARGBHalf))
            throw new Exception("Requires Metal and ARGBHalf RenderTexture");
        var c = new Context();
        try
        {
            c.cameraObject = new GameObject("NBFX Fresnel VFX Camera");
            c.camera = c.cameraObject.AddComponent<Camera>();
            c.camera.orthographic = true;
            c.camera.orthographicSize = 0.5f;
            c.camera.nearClipPlane = 0.1f;
            c.camera.farClipPlane = 20f;
            c.camera.transform.position = new Vector3(0, 0, 8);
            c.camera.transform.rotation = Quaternion.Euler(0, 180, 0);
            c.camera.clearFlags = CameraClearFlags.SolidColor;
            c.camera.backgroundColor = Color.black;
            c.camera.allowHDR = true;
            c.camera.allowMSAA = false;
            c.camera.cullingMask = 1;
            c.target = new RenderTexture(Size, Size, 24,
                RenderTextureFormat.ARGBHalf, RenderTextureReadWrite.Linear);
            c.target.antiAliasing = 1;
            c.target.Create();
            if (!c.target.IsCreated() || c.target.sRGB)
                throw new Exception("Linear ARGBHalf target failed");
            c.camera.targetTexture = c.target;
            c.readback = new Texture2D(Size, Size, TextureFormat.RGBAHalf, false, true);
            c.effects = new VisualEffect[Names.Length];
            for (int i = 0; i < Names.Length; i++)
                c.effects[i] = CreateEffect(Names[i], assets[i]);
            return c;
        }
        catch { Cleanup(c); throw; }
    }
    static VisualEffect CreateEffect(string name, VisualEffectAsset asset)
    {
        var go = new GameObject("NBFX Fresnel " + name);
        go.layer = 0;
        go.transform.position = new Vector3(0, 0, 2);
        go.transform.localScale = Vector3.one * 4f;
        var effect = go.AddComponent<VisualEffect>();
        effect.resetSeedOnPlay = false;
        effect.startSeed = 12345;
        effect.visualEffectAsset = asset;
        effect.Reinit();
        return effect;
    }
    static Color[] Capture(Context c, string path)
    {
        c.camera.Render();
        var previous = RenderTexture.active;
        try
        {
            RenderTexture.active = c.target;
            c.readback.ReadPixels(new Rect(0, 0, Size, Size), 0, 0);
            c.readback.Apply(false);
            var colors = c.readback.GetPixels();
            var preview = new Texture2D(Size, Size, TextureFormat.RGBA32, false, true);
            try {
                preview.SetPixels(colors);
                preview.Apply(false);
                File.WriteAllBytes(path, preview.EncodeToPNG());
            }
            finally { Destroy(preview); }
            return colors;
        }
        finally { RenderTexture.active = previous; }
    }
    static float MaxRGB(Color a, Color b) => Mathf.Max(
        Mathf.Abs(a.r - b.r), Mathf.Abs(a.g - b.g), Mathf.Abs(a.b - b.b));
    static float MaxRGB(Color[] a, Color[] b)
    {
        float max = 0;
        for (int i = 0; i < a.Length; i++) max = Mathf.Max(max, MaxRGB(a[i], b[i]));
        return max;
    }
    static int Changed(Color[] a, Color[] b, float tolerance)
    {
        int count = 0;
        for (int i = 0; i < a.Length; i++)
            if (MaxRGB(a[i], b[i]) > tolerance) count++;
        return count;
    }
    static int ChangedAt(List<int> roi, Color[] a, Color[] b, float tolerance)
    {
        int count = 0;
        foreach (int i in roi) if (MaxRGB(a[i], b[i]) > tolerance) count++;
        return count;
    }
    static float Quantile(List<float> values, float p)
    {
        if (values.Count == 0) throw new Exception("No safe numeric samples");
        values.Sort();
        float at = p * (values.Count - 1);
        int low = Mathf.FloorToInt(at), high = Mathf.CeilToInt(at);
        return Mathf.Lerp(values[low], values[high], at - low);
    }
    static float RGBDistance(CaseInfo a, CaseInfo b) => Mathf.Max(
        Mathf.Abs(a.sourceRMedian - b.sourceRMedian),
        Mathf.Abs(a.sourceGMedian - b.sourceGMedian),
        Mathf.Abs(a.sourceBMedian - b.sourceBMedian));

    static CaseInfo Describe(int index, int alive, bool culled, Shader shader,
        Color[] black, Color[] white, Color[] frameBlack, Color[] frameWhite,
        Color[] repeatBlack, Color[] repeatWhite, List<int> roi, Color[] offBlack)
    {
        var material = new Material(shader);
        try
        {
            var info = new CaseInfo {
                name = Names[index],
                alive = alive,
                culled = culled,
                passCount = material.passCount,
                opaquePass = material.FindPass("NBCameraOpaqueDistortPass"),
                deferredPass = material.FindPass("NBDeferredDistortPass"),
                visiblePixels = Changed(black, frameBlack, 0.02f),
                changedVsOffPixels = ChangedAt(roi, offBlack, frameBlack, 0.04f),
                repeatMaxRGB = Mathf.Max(MaxRGB(frameBlack, repeatBlack),
                                         MaxRGB(frameWhite, repeatWhite))
            };
            var alphas = new List<float>();
            var reds = new List<float>();
            var greens = new List<float>();
            var blues = new List<float>();
            foreach (int i in roi)
            {
                var b0 = black[i]; var b1 = white[i];
                var f0 = frameBlack[i]; var f1 = frameWhite[i];
                float alphaSum = 0;
                for (int channel = 0; channel < 3; channel++)
                {
                    float denominator = b1[channel] - b0[channel];
                    if (denominator < 0.8f)
                        throw new Exception("Black/white clear delta too small");
                    alphaSum += 1f - (f1[channel] - f0[channel]) / denominator;
                }
                float alpha = alphaSum / 3f;
                if (alpha < 0.01f || alpha > 1.05f)
                    throw new Exception("Unphysical inferred effective alpha: " + alpha);
                alphas.Add(alpha);
                reds.Add((f0.r - (1f - alpha) * b0.r) / alpha);
                greens.Add((f0.g - (1f - alpha) * b0.g) / alpha);
                blues.Add((f0.b - (1f - alpha) * b0.b) / alpha);
            }
            info.effectiveAlphaMedian = Quantile(alphas, 0.5f);
            info.effectiveAlphaP10 = Quantile(alphas, 0.1f);
            info.effectiveAlphaP90 = Quantile(alphas, 0.9f);
            info.sourceRMedian = Quantile(reds, 0.5f);
            info.sourceGMedian = Quantile(greens, 0.5f);
            info.sourceBMedian = Quantile(blues, 0.5f);
            return info;
        }
        finally { Destroy(material); }
    }
    void Run(Context c, string output, Result result)
    {
        result.linearTarget = !c.target.sRGB;
        foreach (var effect in c.effects) effect.gameObject.layer = 31;
        c.camera.cullingMask = 0;
        c.camera.backgroundColor = Color.black;
        var black = Capture(c, Path.Combine(output, "black.png"));
        c.camera.backgroundColor = Color.white;
        var white = Capture(c, Path.Combine(output, "white.png"));
        var blackFrames = new Color[Names.Length][];
        var whiteFrames = new Color[Names.Length][];
        var blackRepeats = new Color[Names.Length][];
        var whiteRepeats = new Color[Names.Length][];
        var alive = new int[Names.Length];
        var culled = new bool[Names.Length];
        for (int i = 0; i < Names.Length; i++)
        {
            c.effects[i].gameObject.layer = 0;
            c.camera.cullingMask = 1;
            string stem = Names[i].ToLowerInvariant();
            c.camera.backgroundColor = Color.black;
            blackFrames[i] = Capture(c, Path.Combine(output, stem + "-black.png"));
            blackRepeats[i] = Capture(c, Path.Combine(output, stem + "-black-repeat.png"));
            c.camera.backgroundColor = Color.white;
            whiteFrames[i] = Capture(c, Path.Combine(output, stem + "-white.png"));
            whiteRepeats[i] = Capture(c, Path.Combine(output, stem + "-white-repeat.png"));
            alive[i] = (int)c.effects[i].GetParticleSystemInfo("Simple Loop").aliveCount;
            culled[i] = c.effects[i].culled;
            c.effects[i].gameObject.layer = 31;
        }
        var roi = new List<int>();
        for (int y = 1; y < Size - 1; y++) for (int x = 1; x < Size - 1; x++)
        {
            bool safe = true;
            for (int dy = -1; dy <= 1 && safe; dy++)
                for (int dx = -1; dx <= 1 && safe; dx++)
                {
                    int n = (y + dy) * Size + x + dx;
                    if (MaxRGB(black[n], blackFrames[0][n]) < 0.08f ||
                        MaxRGB(white[n], whiteFrames[0][n]) < 0.08f)
                        safe = false;
                }
            if (safe) roi.Add(y * Size + x);
        }
        result.roiPixels = roi.Count;
        if (roi.Count < 100)
            throw new Exception("Not enough safe particle pixels: " + roi.Count);
        var shaders = new[] { offShader, colorShader, alphaModeShader,
            invertShader, affectAlphaShader, rotationShader };
        var infos = new CaseInfo[Names.Length];
        for (int i = 0; i < Names.Length; i++)
            infos[i] = Describe(i, alive[i], culled[i], shaders[i], black, white,
                blackFrames[i], whiteFrames[i], blackRepeats[i], whiteRepeats[i],
                roi, blackFrames[0]);
        result.off = infos[0]; result.color = infos[1];
        result.alphaMode = infos[2]; result.invert = infos[3];
        result.affectAlpha = infos[4]; result.rotation = infos[5];
        float maxBlack = 0, minWhite = float.MaxValue;
        foreach (int i in roi)
        {
            maxBlack = Mathf.Max(maxBlack, black[i].r, black[i].g, black[i].b);
            minWhite = Mathf.Min(minWhite, white[i].r, white[i].g, white[i].b);
        }
        result.blackBaselineMaxRGB = maxBlack;
        result.whiteBaselineMinRGB = minWhite;
    }
    static bool FixtureValid(Result result)
    {
        if (!result.linearTarget || result.roiPixels < 100 ||
            result.blackBaselineMaxRGB > 0.05f || result.whiteBaselineMinRGB < 0.9f)
            return false;
        foreach (var c in new[] { result.off, result.color, result.alphaMode,
            result.invert, result.affectAlpha, result.rotation })
            if (c == null || c.alive != 1 || c.culled || c.visiblePixels <= 100 ||
                c.repeatMaxRGB > RepeatTolerance || c.passCount < 3 ||
                c.opaquePass != 1 || c.deferredPass != 2)
                return false;
        return true;
    }
    static bool NumericPass(Result r)
    {
        float expectedOffAlpha = 1f - (1f - BaseAlpha) * (1f - BaseAlpha);
        bool off = Mathf.Abs(r.off.effectiveAlphaMedian - expectedOffAlpha) < 0.06f &&
            Mathf.Abs(r.off.sourceRMedian - BaseRGB.x) < 0.06f &&
            Mathf.Abs(r.off.sourceGMedian - BaseRGB.y) < 0.06f &&
            Mathf.Abs(r.off.sourceBMedian - BaseRGB.z) < 0.06f;
        bool color = r.color.sourceRMedian > r.off.sourceRMedian + 0.2f &&
            r.color.sourceGMedian < r.off.sourceGMedian - 0.1f &&
            r.color.sourceBMedian < r.off.sourceBMedian - 0.1f &&
            r.color.effectiveAlphaMedian > r.off.effectiveAlphaMedian + 0.1f;
        bool alpha = RGBDistance(r.alphaMode, r.off) < 0.06f &&
            r.alphaMode.effectiveAlphaMedian < r.off.effectiveAlphaMedian - 0.04f;
        bool invert = r.invert.sourceRMedian < r.color.sourceRMedian - 0.15f &&
            r.invert.effectiveAlphaMedian < r.color.effectiveAlphaMedian - 0.08f;
        bool affect = RGBDistance(r.affectAlpha, r.color) < 0.06f &&
            Mathf.Abs(r.affectAlpha.effectiveAlphaMedian - r.off.effectiveAlphaMedian) < 0.06f &&
            r.color.effectiveAlphaMedian > r.affectAlpha.effectiveAlphaMedian + 0.1f;
        bool rotation = r.rotation.sourceRMedian > r.color.sourceRMedian + 0.08f &&
            r.rotation.effectiveAlphaMedian > r.color.effectiveAlphaMedian + 0.025f;
        return off && color && alpha && invert && affect && rotation &&
            r.color.changedVsOffPixels >= 100 &&
            r.alphaMode.changedVsOffPixels >= 100 &&
            r.invert.changedVsOffPixels >= 100 &&
            r.affectAlpha.changedVsOffPixels >= 100 &&
            r.rotation.changedVsOffPixels >= 100;
    }
    static void Cleanup(Context c)
    {
        if (c == null) return;
        if (c.camera) c.camera.targetTexture = null;
        if (c.target) { c.target.Release(); Destroy(c.target); }
        if (c.readback) Destroy(c.readback);
        if (c.cameraObject) Destroy(c.cameraObject);
        if (c.effects != null) foreach (var effect in c.effects)
            if (effect) Destroy(effect.gameObject);
    }
}
