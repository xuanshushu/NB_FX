// Test-only isolated-clone Editor harness. Never copy to the NB_FX package.
using System;
using System.IO;
using System.Reflection;
using System.Text.RegularExpressions;
using UnityEditor;
using UnityEditor.Build.Reporting;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.VFX;

public static class NBFXBackPlayerBuild
{
    const string Root = "/tmp/nbfx-vfx-backcolor-probe-20260930/run";
    const string QuadPath = "Assets/NBFXBackQuad.asset";
    const string ScenePath = "Assets/NBFXBackPlayer.unity";
    const string AppPath = Root + "/player/NBFXBackPlayer.app";
    const string GraphPath = "Packages/com.xuanxuan.nb.fx/NBShaders2/ShaderGraph/NBShaderGraph.shadergraph";
    const string WhiteTexturePath = "Assets/NBFXBCWhite.png";
    const string WhiteTextureGuid = "b2d9b4eef3a75cec96533624937d9a16";
    const string CubeReference = "{\"obj\":{\"fileID\":10202,\"guid\":\"0000000000000000e000000000000000\",\"type\":0}}";
    const long OutputId = 8926484042661614853;
    const long MeshId = 8926484042661614862;
    const long BaseMapId = 8926484042661614854;
    const long FlagsHiId = 8926484042661614870;
    const long BackColorId = 8926484042661616002;
    static readonly string[] Paths = { "Assets/NBFXBackOff.vfx", "Assets/NBFXBackOn.vfx" };
    static readonly string[] Guids = {
        "1485fc93dfa25f70bee1f1ea7fab31af",
        "5b20ad8671be5877aaebb953c9beb137"
    };
    static readonly int[] FlagsHi = { 0, 4096 };

    public static void Run()
    {
        Directory.CreateDirectory(Root);
        using (var report = new StreamWriter(Path.Combine(Root, "editor-inspection.txt")))
        {
            ValidateWhiteTexture(report);
            var quad = CreateOrValidateQuad(report);
            if (!AssetDatabase.TryGetGUIDAndLocalFileIdentifier(quad, out string meshGuid, out long meshId))
                throw new Exception("Could not identify clone-only Quad Mesh asset");
            string reference = "{\"obj\":{\"fileID\":" + meshId + ",\"guid\":\"" + meshGuid + "\",\"type\":2}}";
            report.WriteLine("quadMesh=" + QuadPath + " guid=" + meshGuid + " localFileID=" + meshId);
            var graph = AssetDatabase.LoadAssetAtPath<Shader>(GraphPath);
            if (!graph || !graph.isSupported) throw new Exception("Product Graph unavailable");
            var assets = new VisualEffectAsset[Paths.Length];
            var shaders = new Shader[Paths.Length];
            for (int i = 0; i < Paths.Length; i++)
            {
                PatchTestMeshReference(Paths[i], reference, report);
                AssetDatabase.ImportAsset(Paths[i], ImportAssetOptions.ForceUpdate);
                if (AssetDatabase.AssetPathToGUID(Paths[i]) != Guids[i])
                    throw new Exception("VFX fixture GUID changed: " + Paths[i]);
                ValidateSerialized(Paths[i], reference, FlagsHi[i], report);
                assets[i] = AssetDatabase.LoadAssetAtPath<VisualEffectAsset>(Paths[i]);
                shaders[i] = FindShader(Paths[i]);
                if (!assets[i] || !shaders[i] || !shaders[i].isSupported || !shaders[i].name.Contains("Mesh"))
                    throw new Exception("Real VFX Mesh Output failed to import: " + Paths[i]);
                ValidateGeneratedShader(Paths[i], FlagsHi[i], report);
                var material = new Material(shaders[i]);
                try
                {
                    report.WriteLine(Paths[i] + " shader=" + shaders[i].name +
                        " lazyEditorPassCount=" + material.passCount);
                }
                finally { UnityEngine.Object.DestroyImmediate(material); }
            }
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            var host = new GameObject("NBFX BackColor Player Probe");
            var probe = host.AddComponent<NBFXBackPlayerProbe>();
            probe.offAsset = assets[0]; probe.onAsset = assets[1];
            probe.offShader = shaders[0]; probe.onShader = shaders[1];
            probe.graphShader = graph; probe.quadMesh = quad;
            if (!EditorSceneManager.SaveScene(scene, ScenePath))
                throw new Exception("Could not save clone-only Player scene");
            report.Flush();
        }
        Directory.CreateDirectory(Path.GetDirectoryName(AppPath));
        var build = BuildPipeline.BuildPlayer(new BuildPlayerOptions
        {
            scenes = new[] { ScenePath }, locationPathName = AppPath,
            target = BuildTarget.StandaloneOSX, options = BuildOptions.None
        });
        Debug.Log("NBFX_BACK_BUILD result=" + build.summary.result +
            " errors=" + build.summary.totalErrors + " warnings=" + build.summary.totalWarnings);
        if (build.summary.result != BuildResult.Succeeded || build.summary.totalErrors != 0)
            throw new Exception("BackColor VFX Player build failed");
    }

    static void ValidateWhiteTexture(TextWriter report)
    {
        AssetDatabase.ImportAsset(WhiteTexturePath, ImportAssetOptions.ForceUpdate);
        var importer = AssetImporter.GetAtPath(WhiteTexturePath) as TextureImporter;
        var texture = AssetDatabase.LoadAssetAtPath<Texture2D>(WhiteTexturePath);
        if (!importer || !texture || AssetDatabase.AssetPathToGUID(WhiteTexturePath) != WhiteTextureGuid ||
            importer.sRGBTexture || importer.mipmapEnabled || !importer.isReadable ||
            importer.textureCompression != TextureImporterCompression.Uncompressed ||
            texture.width != 1 || texture.height != 1 ||
            importer.filterMode != FilterMode.Point || importer.wrapMode != TextureWrapMode.Clamp)
            throw new Exception("White BaseMap texture import contract failed");
        var pixel = texture.GetPixel(0, 0);
        for (int i = 0; i < 4; i++)
            if (Mathf.Abs(pixel[i] - 1f) > 0.003f)
                throw new Exception("White BaseMap pixel is not (1,1,1,1): " + pixel);
        report.WriteLine("BaseMap=explicit white RGBA 1x1, linear, point/clamp, uncompressed");
    }

    static Mesh CreateOrValidateQuad(TextWriter report)
    {
        var mesh = AssetDatabase.LoadAssetAtPath<Mesh>(QuadPath);
        if (!mesh)
        {
            mesh = new Mesh { name = "NBFXBackQuad" };
            mesh.vertices = new[] {
                new Vector3(-0.5f, -0.5f, 0), new Vector3(0.5f, -0.5f, 0),
                new Vector3(0.5f, 0.5f, 0), new Vector3(-0.5f, 0.5f, 0) };
            mesh.uv = new[] { Vector2.zero, Vector2.right, Vector2.one, Vector2.up };
            mesh.normals = new[] { Vector3.forward, Vector3.forward, Vector3.forward, Vector3.forward };
            mesh.triangles = new[] { 0, 1, 2, 0, 2, 3 };
            mesh.RecalculateBounds();
            AssetDatabase.CreateAsset(mesh, QuadPath);
            AssetDatabase.SaveAssets();
        }
        var triangles = mesh.triangles;
        if (mesh.vertexCount != 4 || triangles.Length != 6 ||
            triangles[0] != 0 || triangles[1] != 1 || triangles[2] != 2 ||
            triangles[3] != 0 || triangles[4] != 2 || triangles[5] != 3 ||
            mesh.normals.Length != 4 || mesh.normals[0].z < 0.99f)
            throw new Exception("Clone-only Quad is not the expected one-sided +Z mesh");
        report.WriteLine("quad vertexCount=4 triangles=(0,1,2,0,2,3) normals=+Z");
        return mesh;
    }

    static string Body(string source, long fileId)
    {
        var match = Regex.Match(source, @"(?ms)^--- !u!114 &" + fileId + @"\n.*?(?=^--- !u!|\z)");
        if (!match.Success) throw new Exception("Missing VFX object " + fileId);
        return match.Value;
    }

    static void PatchTestMeshReference(string path, string reference, TextWriter report)
    {
        string absolute = Path.Combine(Directory.GetParent(Application.dataPath).FullName, path);
        string text = File.ReadAllText(absolute);
        string body = Body(text, MeshId);
        if (body.Contains(reference))
        {
            report.WriteLine(path + " already references clone Quad Mesh");
            return;
        }
        if (!body.Contains(CubeReference))
            throw new Exception("Unknown test-only VFX Mesh slot schema; stop before import: " + path);
        body = body.Replace(CubeReference, reference);
        var match = Regex.Match(text, @"(?ms)^--- !u!114 &" + MeshId + @"\n.*?(?=^--- !u!|\z)");
        text = text.Substring(0, match.Index) + body + text.Substring(match.Index + match.Length);
        File.WriteAllText(absolute, text);
        report.WriteLine(path + " patched only its test Output Mesh slot to clone Quad");
    }

    static void ValidateSerialized(string path, string meshReference, int flag, TextWriter report)
    {
        string text = File.ReadAllText(Path.Combine(Directory.GetParent(Application.dataPath).FullName, path));
        string output = Body(text, OutputId), back = Body(text, BackColorId);
        if (!text.Contains("  capacity: 1\n") || !output.Contains("  cullMode: 0\n") ||
            !output.Contains("  blendMode: 1\n") ||
            !output.Contains("  - {fileID: " + BackColorId + "}\n") ||
            !Body(text, MeshId).Contains(meshReference) ||
            !Body(text, BaseMapId).Contains(WhiteTextureGuid) ||
            !Body(text, FlagsHiId).Contains("name: _NB_Flags0Hi16") ||
            !Body(text, FlagsHiId).Contains("m_SerializableObject: " + flag + "\n") ||
            !back.Contains("name: _BaseBackColor") ||
            !back.Contains("{\"r\":1.0,\"g\":0.0,\"b\":0.0,\"a\":1.0}"))
            throw new Exception("BackColor VFX Output serialized contract invalid: " + path);
        report.WriteLine(path + " flagHi16=" + flag + " BackColor=red CullOff QuadMesh");
    }

    static Shader FindShader(string path)
    {
        foreach (var obj in AssetDatabase.LoadAllAssetsAtPath(path))
            if (obj is Shader shader && shader.name.Contains("Mesh")) return shader;
        return null;
    }

    static void ValidateGeneratedShader(string path, int flag, TextWriter report)
    {
        const BindingFlags flags = BindingFlags.Public | BindingFlags.NonPublic |
            BindingFlags.Static | BindingFlags.Instance;
        Type type = null;
        foreach (var assembly in AppDomain.CurrentDomain.GetAssemblies())
        { type = assembly.GetType("UnityEditor.VFX.VisualEffectResource"); if (type != null) break; }
        if (type == null) throw new Exception("VFX generated-resource reflection unavailable");
        var resource = type.GetMethod("GetResourceAtPath", flags).Invoke(null, new object[] { path });
        if (resource == null) throw new Exception("VFX generated-resource missing");
        int count = (int)type.GetMethod("GetShaderSourceCount", flags).Invoke(resource, null);
        var getter = type.GetMethod("GetShaderSource", flags);
        bool found = false;
        for (int i = 0; i < count; i++)
        {
            string source = (string)getter.Invoke(resource, new object[] { i });
            if (!source.Contains("output._BaseBackColor")) continue;
            File.WriteAllText(Path.Combine(Root,
                Path.GetFileNameWithoutExtension(path) + "-" + i + ".shader"), source);
            bool values = source.Contains("output._NB_Flags0Hi16 = (float)" + flag + ";") &&
                Regex.IsMatch(source,
                    @"output\._BaseBackColor\s*=\s*float4\(\s*1(?:\.0)?\s*,\s*0(?:\.0)?\s*,\s*0(?:\.0)?\s*,\s*1(?:\.0)?\s*\)");
            bool facesAndPasses = source.Contains("FaceSign") &&
                source.Contains("\"LightMode\" = \"NBCameraOpaqueDistortPass\"") &&
                source.Contains("\"LightMode\" = \"NBDeferredDistortPass\"");
            report.WriteLine(path + " generatedSource=" + i + " redAndFlag=" + values +
                " FaceSignAndPasses=" + facesAndPasses);
            found |= values && facesAndPasses;
        }
        if (!found) throw new Exception("BackColor/Flags0 Output binding or exact custom Pass missing: " + path);
    }
}
