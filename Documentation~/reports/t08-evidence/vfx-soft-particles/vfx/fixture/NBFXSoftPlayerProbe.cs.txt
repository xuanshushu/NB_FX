// /tmp-only real one-particle VFX Mesh Soft Particles Player probe.
using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;
using UnityEngine.VFX;

public sealed class NBFXSoftPlayerProbe : MonoBehaviour
{
    public VisualEffectAsset offAsset, onAsset;
    public Shader graphShader, offShader, onShader;
    public Material boardMaterialTemplate;
    const int Size = 256;
    const float RepeatTolerance = .005f;
    const float PixelTolerance = .02f;
    const float CameraZ = 7f, EffectZ = 2f, BoardZ = 1.5f;
    static readonly Color BoardColor = new Color(.08f, .08f, .08f, 1f);

    [Serializable] sealed class CaseInfo
    {
        public string name;
        public int alive, passCount, opaquePass, deferredPass, visiblePixels;
        public bool culled;
        public float repeatMaxRGB, medianSignal, relativeToOff;
    }
    [Serializable] sealed class Condition
    {
        public string name;
        public bool boardVisible;
        public int roiPixels;
        public float baselineBlackWhiteMaxRGB, offOnMaxRGB;
        public CaseInfo off, on;
    }
    [Serializable] sealed class Result
    {
        public string unityVersion, graphicsDevice, status, error, note;
        public bool linearTarget, requiresDepthTexture, fixtureValid, pass;
        public float cameraZ, effectZ, boardZ;
        public Condition empty, occluded;
    }
    sealed class Context
    {
        public GameObject cameraObject, boardObject;
        public Camera camera;
        public RenderTexture target;
        public Texture2D readback;
        public Material boardMaterial;
        public VisualEffect[] effects;
    }
    sealed class Frames
    {
        public Color[] baselineBlack, baselineWhite;
        public Color[][] black = new Color[2][], white = new Color[2][];
        public Color[][] repeatBlack = new Color[2][], repeatWhite = new Color[2][];
        public int[] alive = new int[2];
        public bool[] culled = new bool[2];
    }
    static string OutputPath()
    {
        foreach (string arg in Environment.GetCommandLineArgs())
            if (arg.StartsWith("--nbfx-soft-output=", StringComparison.Ordinal))
                return arg.Substring("--nbfx-soft-output=".Length);
        return null;
    }
    IEnumerator Start()
    {
        string output = OutputPath();
        if (string.IsNullOrEmpty(output)) { Application.Quit(2); yield break; }
        Directory.CreateDirectory(output);
        var result = new Result {
            unityVersion = Application.unityVersion,
            graphicsDevice = SystemInfo.graphicsDeviceType.ToString(),
            cameraZ = CameraZ, effectZ = EffectZ, boardZ = BoardZ,
            note = "Real one-particle Cube/CullOff VFX Mesh Output. Off/On differ only in _SoftParticlesEnabled; both use (near,far)=(0,1). URP camera requiresDepthTexture=true. Fixed opaque URP Unlit board at z=1.5 is behind effect center z=2; empty is negative control. On-board metric is source-color displacement from the SAME board baseline, normalized by Off, not a direct fragment/GPU alpha."
        };
        Context c = null;
        int exitCode = 2;
        try { c = Setup(); }
        catch (Exception ex) { result.status = "FIXTURE_INVALID"; result.error = ex.ToString(); }
        if (c != null && result.error == null)
        {
            // Same deterministic one-particle warmup/pause schedule as Distance Fade.
            for (int frame = 0; frame < 3; frame++) { c.camera.Render(); yield return null; }
            foreach (var effect in c.effects) effect.Simulate(1f / 60f, 21u);
            for (int frame = 0; frame < 2; frame++) { c.camera.Render(); yield return null; }
            foreach (var effect in c.effects) effect.pause = true;
            try {
                var empty = CaptureFrames(c, output, "empty", false);
                var occluded = CaptureFrames(c, output, "occluded", true);
                var roi = BuildROI(empty);
                result.empty = Describe("empty", false, empty, roi, offShader, onShader);
                result.occluded = Describe("occluded", true, occluded, roi, offShader, onShader);
                result.linearTarget = !c.target.sRGB;
                result.requiresDepthTexture = c.camera.GetUniversalAdditionalCameraData().requiresDepthTexture;
                result.fixtureValid = FixtureValid(result);
                result.pass = result.fixtureValid && NumericPass(result);
                result.status = result.pass ? "PASS" : result.fixtureValid ? "NUMERIC_MISMATCH" : "FIXTURE_INVALID";
                exitCode = result.pass ? 0 : 1;
            }
            catch (Exception ex) { result.status = "FIXTURE_INVALID"; result.error = ex.ToString(); }
        }
        try { Cleanup(c); }
        catch (Exception ex) { result.status = "CLEANUP_FAILED"; result.error = (result.error ?? "") + "\n" + ex; exitCode = 2; }
        File.WriteAllText(Path.Combine(output, "result.json"), JsonUtility.ToJson(result, true));
        Debug.Log("NBFX_SOFT_PLAYER " + JsonUtility.ToJson(result));
        Application.Quit(exitCode);
    }
    Context Setup()
    {
        if (!graphShader || !offAsset || !onAsset || !offShader || !onShader || !boardMaterialTemplate)
            throw new Exception("Graph/VFX/Shader reference missing");
        if (SystemInfo.graphicsDeviceType != GraphicsDeviceType.Metal ||
            !SystemInfo.SupportsRenderTextureFormat(RenderTextureFormat.ARGBHalf))
            throw new Exception("Requires Metal and ARGBHalf RenderTexture");
        var c = new Context();
        try {
            c.cameraObject = new GameObject("NBFX Soft Particles VFX Camera");
            c.camera = c.cameraObject.AddComponent<Camera>();
            c.camera.orthographic = true;
            c.camera.orthographicSize = .5f;
            c.camera.nearClipPlane = .1f;
            c.camera.farClipPlane = 20f;
            c.camera.transform.position = new Vector3(0, 0, CameraZ);
            c.camera.transform.rotation = Quaternion.Euler(0, 180, 0);
            c.camera.clearFlags = CameraClearFlags.SolidColor;
            c.camera.backgroundColor = Color.black;
            c.camera.allowHDR = true;
            c.camera.allowMSAA = false;
            c.camera.cullingMask = 1;
            c.camera.GetUniversalAdditionalCameraData().requiresDepthTexture = true;
            c.target = new RenderTexture(Size, Size, 24, RenderTextureFormat.ARGBHalf,
                RenderTextureReadWrite.Linear);
            c.target.antiAliasing = 1;
            c.target.Create();
            if (!c.target.IsCreated() || c.target.sRGB)
                throw new Exception("Linear ARGBHalf target failed");
            c.camera.targetTexture = c.target;
            c.readback = new Texture2D(Size, Size, TextureFormat.RGBAHalf, false, true);
            if (!boardMaterialTemplate.shader || !boardMaterialTemplate.shader.isSupported)
                throw new Exception("URP Unlit board material unavailable");
            c.boardMaterial = new Material(boardMaterialTemplate);
            c.boardMaterial.SetColor("_BaseColor", BoardColor);
            c.boardMaterial.SetFloat("_Surface", 0f);
            c.boardMaterial.SetFloat("_ZWrite", 1f);
            c.boardMaterial.SetFloat("_Cull", 0f);
            c.boardMaterial.renderQueue = 2000;
            c.boardObject = new GameObject("NBFX fixed-color opaque depth board");
            var boardMesh = new Mesh();
            boardMesh.vertices = new[] {
                new Vector3(-1f,-1f,0f), new Vector3(1f,-1f,0f),
                new Vector3(-1f,1f,0f), new Vector3(1f,1f,0f)
            };
            boardMesh.triangles = new[] { 0,1,2,2,1,3 };
            boardMesh.RecalculateNormals(); boardMesh.RecalculateBounds();
            c.boardObject.AddComponent<MeshFilter>().sharedMesh = boardMesh;
            c.boardObject.AddComponent<MeshRenderer>();
            c.boardObject.transform.position = new Vector3(0, 0, BoardZ);
            c.boardObject.transform.localScale = new Vector3(2f, 2f, 1f);
            c.boardObject.GetComponent<Renderer>().sharedMaterial = c.boardMaterial;
            Debug.Log("NBFX_SOFT_BOARD setup mesh=" + c.boardObject.GetComponent<MeshFilter>().sharedMesh.vertexCount + " mat=" + c.boardMaterial.shader.name + " passCount=" + c.boardMaterial.passCount + " cameraMask=" + c.camera.cullingMask);
            c.boardObject.layer = 0;
            c.boardObject.SetActive(false);
            c.effects = new VisualEffect[2];
            c.effects[0] = CreateEffect("Off", offAsset);
            c.effects[1] = CreateEffect("On", onAsset);
            return c;
        }
        catch { Cleanup(c); throw; }
    }
    static VisualEffect CreateEffect(string name, VisualEffectAsset asset)
    {
        var go = new GameObject("NBFX Soft Particles " + name);
        go.layer = 0;
        go.transform.position = new Vector3(0, 0, EffectZ);
        go.transform.localScale = Vector3.one * 4f;
        var effect = go.AddComponent<VisualEffect>();
        effect.resetSeedOnPlay = false;
        effect.startSeed = 12345;
        effect.visualEffectAsset = asset;
        effect.Reinit();
        return effect;
    }
    static Color[] Capture(Context c, string path)
    {
        c.camera.Render();
        var previous = RenderTexture.active;
        try {
            RenderTexture.active = c.target;
            c.readback.ReadPixels(new Rect(0, 0, Size, Size), 0, 0);
            c.readback.Apply(false);
            var colors = c.readback.GetPixels();
            var preview = new Texture2D(Size, Size, TextureFormat.RGBA32, false, true);
            try {
                preview.SetPixels(colors);
                preview.Apply(false);
                File.WriteAllBytes(path, preview.EncodeToPNG());
            }
            finally { Destroy(preview); }
            return colors;
        }
        finally { RenderTexture.active = previous; }
    }
    static Frames CaptureFrames(Context c, string output, string name, bool board)
    {
        var f = new Frames();
        foreach (var e in c.effects) e.gameObject.layer = 31;
        c.boardObject.SetActive(board);
        Debug.Log("NBFX_SOFT_BOARD case=" + name + " active=" + c.boardObject.activeSelf + " layer=" + c.boardObject.layer + " bounds=" + c.boardObject.GetComponent<Renderer>().bounds);
        c.camera.backgroundColor = Color.black;
        f.baselineBlack = Capture(c, Path.Combine(output, name + "-baseline-black.png"));
        c.camera.backgroundColor = Color.white;
        f.baselineWhite = Capture(c, Path.Combine(output, name + "-baseline-white.png"));
        for (int i = 0; i < 2; i++) {
            c.effects[i].gameObject.layer = 0;
            string stem = name + (i == 0 ? "-off" : "-on");
            c.camera.backgroundColor = Color.black;
            f.black[i] = Capture(c, Path.Combine(output, stem + "-black.png"));
            f.repeatBlack[i] = Capture(c, Path.Combine(output, stem + "-black-repeat.png"));
            c.camera.backgroundColor = Color.white;
            f.white[i] = Capture(c, Path.Combine(output, stem + "-white.png"));
            f.repeatWhite[i] = Capture(c, Path.Combine(output, stem + "-white-repeat.png"));
            f.alive[i] = (int)c.effects[i].GetParticleSystemInfo("Simple Loop").aliveCount;
            f.culled[i] = c.effects[i].culled;
            c.effects[i].gameObject.layer = 31;
        }
        return f;
    }
    static float MaxRGB(Color a, Color b) => Mathf.Max(
        Mathf.Abs(a.r - b.r), Mathf.Abs(a.g - b.g), Mathf.Abs(a.b - b.b));
    static float MaxRGB(Color[] a, Color[] b)
    {
        float max = 0f;
        for (int i = 0; i < a.Length; i++) max = Mathf.Max(max, MaxRGB(a[i], b[i]));
        return max;
    }
    static int Changed(Color[] a, Color[] b, float tolerance)
    {
        int count = 0;
        for (int i = 0; i < a.Length; i++)
            if (MaxRGB(a[i], b[i]) > tolerance) count++;
        return count;
    }
    static float Quantile(List<float> values, float p)
    {
        if (values.Count == 0) throw new Exception("No safe numeric samples");
        values.Sort();
        float at = p * (values.Count - 1);
        int low = Mathf.FloorToInt(at), high = Mathf.CeilToInt(at);
        return Mathf.Lerp(values[low], values[high], at - low);
    }
    static List<int> BuildROI(Frames empty)
    {
        var roi = new List<int>();
        for (int y = 1; y < Size - 1; y++) for (int x = 1; x < Size - 1; x++) {
            bool safe = true;
            for (int dy = -1; dy <= 1 && safe; dy++)
                for (int dx = -1; dx <= 1 && safe; dx++) {
                    int n = (y + dy) * Size + x + dx;
                    if (MaxRGB(empty.baselineBlack[n], empty.black[0][n]) < .08f ||
                        MaxRGB(empty.baselineWhite[n], empty.white[0][n]) < .08f)
                        safe = false;
                }
            if (safe) roi.Add(y * Size + x);
        }
        if (roi.Count < 100) throw new Exception("Too few safe no-board particle pixels: " + roi.Count);
        return roi;
    }
    static float Signal(Color frame, Color baseline)
    {
        var d = new Vector3(frame.r - baseline.r, frame.g - baseline.g, frame.b - baseline.b);
        return d.magnitude;
    }
    static float RelativeSignal(Color frame, Color baseline, Color off)
    {
        var d = new Vector3(off.r - baseline.r, off.g - baseline.g, off.b - baseline.b);
        var q = new Vector3(frame.r - baseline.r, frame.g - baseline.g, frame.b - baseline.b);
        float denom = Vector3.Dot(d, d);
        if (denom < .0025f) throw new Exception("Opaque-board Off signal too small");
        return Vector3.Dot(q, d) / denom;
    }
    static Condition Describe(string name, bool board, Frames f, List<int> roi,
        Shader offShader, Shader onShader)
    {
        var result = new Condition {
            name = name, boardVisible = board, roiPixels = roi.Count,
            baselineBlackWhiteMaxRGB = MaxRGB(f.baselineBlack, f.baselineWhite)
        };
        float offOnMax = 0;
        foreach (int k in roi)
            offOnMax = Mathf.Max(offOnMax, MaxRGB(f.black[0][k], f.black[1][k]));
        result.offOnMaxRGB = offOnMax;
        for (int i = 0; i < 2; i++) {
            var mat = new Material(i == 0 ? offShader : onShader);
            try {
                var info = new CaseInfo {
                    name = name + (i == 0 ? "Off" : "On"),
                    alive = f.alive[i], culled = f.culled[i],
                    passCount = mat.passCount,
                    opaquePass = mat.FindPass("NBCameraOpaqueDistortPass"),
                    deferredPass = mat.FindPass("NBDeferredDistortPass"),
                    visiblePixels = Changed(f.baselineBlack, f.black[i], PixelTolerance),
                    repeatMaxRGB = Mathf.Max(MaxRGB(f.black[i], f.repeatBlack[i]),
                        MaxRGB(f.white[i], f.repeatWhite[i]))
                };
                var signals = new List<float>();
                var ratios = new List<float>();
                foreach (int k in roi) {
                    signals.Add(Signal(f.black[i][k], f.baselineBlack[k]));
                    if (board) ratios.Add(RelativeSignal(f.black[i][k],
                        f.baselineBlack[k], f.black[0][k]));
                }
                info.medianSignal = Quantile(signals, .5f);
                info.relativeToOff = board ? Quantile(ratios, .5f) : 0f;
                if (i == 0) result.off = info; else result.on = info;
            }
            finally { Destroy(mat); }
        }
        return result;
    }
    static bool FixtureValid(Result r)
    {
        if (!r.linearTarget || !r.requiresDepthTexture || r.empty == null ||
            r.occluded == null || r.empty.roiPixels < 100 ||
            r.occluded.roiPixels != r.empty.roiPixels ||
            r.empty.baselineBlackWhiteMaxRGB < .9f ||
            r.occluded.baselineBlackWhiteMaxRGB > .01f ||
            r.occluded.off.medianSignal < .08f)
            return false;
        foreach (var condition in new[] { r.empty, r.occluded })
            foreach (var c in new[] { condition.off, condition.on })
                if (c == null || c.alive != 1 || c.culled || c.visiblePixels < 30 ||
                    c.repeatMaxRGB > RepeatTolerance || c.passCount < 3 ||
                    c.opaquePass != 1 || c.deferredPass != 2)
                    return false;
        return true;
    }
    static bool NumericPass(Result r)
    {
        // Empty-depth negative control: On must not fade without opaque geometry.
        bool empty = r.empty.offOnMaxRGB < .01f &&
            Mathf.Abs(r.empty.off.medianSignal - r.empty.on.medianSignal) < .01f;
        // Board positive control: the fixed-color opaque board creates depth.
        // Relative signal compares the particle contribution against the same
        // board baseline, not black/white-based alpha (invalid over opaque board).
        bool board = r.occluded.offOnMaxRGB > .03f &&
            r.occluded.off.relativeToOff > .98f &&
            r.occluded.off.relativeToOff < 1.02f &&
            r.occluded.on.relativeToOff > .05f &&
            r.occluded.on.relativeToOff < .85f &&
            r.occluded.on.medianSignal < r.occluded.off.medianSignal - .03f;
        return empty && board;
    }
    static void Cleanup(Context c)
    {
        if (c == null) return;
        if (c.camera) c.camera.targetTexture = null;
        if (c.target) { c.target.Release(); Destroy(c.target); }
        if (c.readback) Destroy(c.readback);
        if (c.boardObject) { var mesh = c.boardObject.GetComponent<MeshFilter>(); if (mesh && mesh.sharedMesh) Destroy(mesh.sharedMesh); Destroy(c.boardObject); }
        if (c.boardMaterial) Destroy(c.boardMaterial);
        if (c.cameraObject) Destroy(c.cameraObject);
        if (c.effects != null) foreach (var e in c.effects)
            if (e) Destroy(e.gameObject);
    }
}
