using System;
using System.Collections;
using System.IO;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.VFX;

public sealed class NBFXT08AgePlayerProbe : MonoBehaviour
{
    public VisualEffectAsset constantAsset,ageAsset;
    public Shader graphShader,constantShader,ageShader;
    const int Size=128;
    [Serializable] sealed class Result
    {
        public string unityVersion,graphicsDevice,error;
        public int aliveConstant,aliveAge,visibleConstant,visibleAge,changedPair,constantPasses,agePasses;
        public bool culledConstant,culledAge,passed;
    }
    static Color32[] Capture(Camera cam,RenderTexture target,Texture2D read,string path)
    {
        cam.Render();var prior=RenderTexture.active;RenderTexture.active=target;
        try {read.ReadPixels(new Rect(0,0,Size,Size),0,0);read.Apply(false);File.WriteAllBytes(path,read.EncodeToPNG());return read.GetPixels32();}
        finally{RenderTexture.active=prior;}
    }
    static int Changed(Color32[] a,Color32[] b)
    {
        int count=0;for(int i=0;i<a.Length;i++)if(Math.Abs(a[i].r-b[i].r)>2||Math.Abs(a[i].g-b[i].g)>2||Math.Abs(a[i].b-b[i].b)>2)count++;return count;
    }
    IEnumerator Start()
    {
        string output=null;foreach(var arg in Environment.GetCommandLineArgs())if(arg.StartsWith("--nbfx-age-output=",StringComparison.Ordinal))output=arg.Substring("--nbfx-age-output=".Length);
        if(string.IsNullOrEmpty(output))yield break;Directory.CreateDirectory(output);
        var result=new Result{unityVersion=Application.unityVersion,graphicsDevice=SystemInfo.graphicsDeviceType.ToString()};
        RenderTexture rt=null;Texture2D read=null,gradient=null;Material bgMat=null,constMat=null,ageMat=null;GameObject bg=null,camGo=null,constGo=null,ageGo=null;
        try
        {
            if(!constantAsset||!ageAsset||!graphShader||!constantShader||!ageShader)throw new Exception("Serialized assets absent");
            gradient=new Texture2D(Size,Size,TextureFormat.RGBA32,false,true);var px=new Color32[Size*Size];
            for(int y=0;y<Size;y++)for(int x=0;x<Size;x++)px[y*Size+x]=new Color32((byte)(x*255/127),(byte)(y*255/127),50,255);
            gradient.SetPixels32(px);gradient.Apply(false);gradient.filterMode=FilterMode.Point;
            bgMat=new Material(graphShader);bgMat.SetTexture("_BaseMap",gradient);bgMat.SetColor("_Color",Color.white);
            bgMat.SetFloat("_Surface",0);bgMat.SetFloat("_SrcBlend",(float)BlendMode.One);bgMat.SetFloat("_DstBlend",(float)BlendMode.Zero);bgMat.SetFloat("_ZWrite",1);bgMat.DisableKeyword("_SURFACE_TYPE_TRANSPARENT");bgMat.renderQueue=2000;
            bg=GameObject.CreatePrimitive(PrimitiveType.Quad);bg.layer=4;bg.transform.position=new Vector3(0,1,1);bg.transform.localScale=new Vector3(6,6,1);bg.GetComponent<MeshRenderer>().sharedMaterial=bgMat;
            camGo=new GameObject("Age Probe Camera");var cam=camGo.AddComponent<Camera>();cam.orthographic=true;cam.orthographicSize=3;cam.nearClipPlane=.1f;cam.farClipPlane=20;cam.transform.position=new Vector3(0,1,8);cam.transform.rotation=Quaternion.Euler(0,180,0);cam.clearFlags=CameraClearFlags.SolidColor;cam.backgroundColor=Color.black;cam.allowHDR=false;cam.allowMSAA=false;
            rt=new RenderTexture(Size,Size,24,RenderTextureFormat.ARGB32);rt.Create();cam.targetTexture=rt;read=new Texture2D(Size,Size,TextureFormat.RGBA32,false,true);
            constGo=new GameObject("Constant1 VFX");constGo.layer=2;constGo.transform.position=new Vector3(0,0,2);var constant=constGo.AddComponent<VisualEffect>();constant.resetSeedOnPlay=false;constant.startSeed=12345;constant.visualEffectAsset=constantAsset;constant.Reinit();
            ageGo=new GameObject("AgeBound VFX");ageGo.layer=3;ageGo.transform.position=new Vector3(0,0,2);var age=ageGo.AddComponent<VisualEffect>();age.resetSeedOnPlay=false;age.startSeed=12345;age.visualEffectAsset=ageAsset;age.Reinit();
            cam.cullingMask=(1<<4)|(1<<2)|(1<<3);for(int i=0;i<20;i++){cam.Render();System.Threading.Thread.Sleep(20);yield return null;}
            result.aliveConstant=(int)constant.GetParticleSystemInfo("Simple Loop").aliveCount;result.aliveAge=(int)age.GetParticleSystemInfo("Simple Loop").aliveCount;result.culledConstant=constant.culled;result.culledAge=age.culled;
            cam.cullingMask=1<<4;var baseline=Capture(cam,rt,read,Path.Combine(output,"background.png"));
            cam.cullingMask=(1<<4)|(1<<2);var a=Capture(cam,rt,read,Path.Combine(output,"constant1.png"));
            cam.cullingMask=(1<<4)|(1<<3);var b=Capture(cam,rt,read,Path.Combine(output,"age_bound.png"));
            result.visibleConstant=Changed(baseline,a);result.visibleAge=Changed(baseline,b);result.changedPair=Changed(a,b);
            constMat=new Material(constantShader);ageMat=new Material(ageShader);result.constantPasses=constMat.passCount;result.agePasses=ageMat.passCount;
            result.passed=result.graphicsDevice=="Metal"&&result.aliveConstant>0&&result.aliveAge>0&&result.visibleConstant>0&&result.visibleAge>0&&result.changedPair>0&&!result.culledConstant&&!result.culledAge&&constMat.FindPass("NBCameraOpaqueDistortPass")==1&&ageMat.FindPass("NBDeferredDistortPass")==2;
        }
        finally
        {
            File.WriteAllText(Path.Combine(output,"result.json"),JsonUtility.ToJson(result,true));Debug.Log("NBFX_T08_AGE_PLAYER "+JsonUtility.ToJson(result));
            if(rt){rt.Release();Destroy(rt);}if(read)Destroy(read);if(gradient)Destroy(gradient);if(bgMat)Destroy(bgMat);if(constMat)Destroy(constMat);if(ageMat)Destroy(ageMat);
            if(bg)Destroy(bg);if(camGo)Destroy(camGo);if(constGo)Destroy(constGo);if(ageGo)Destroy(ageGo);
            Application.Quit(result.passed?0:1);
        }
    }
}
